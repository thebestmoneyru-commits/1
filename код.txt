function normalizeModelId(requestedModel: string): string {
  if (
    !requestedModel ||
    requestedModel === 'gemini-3.8-flash' ||
    requestedModel === 'gemini-flash-latest'
  ) {
    return 'gemini-4-hydrogen';
  }

  return requestedModel;
}

function errorStatus(err: any): number {
  const msg = String(err?.message || '');
  const code = Number(err?.status || err?.code || 0);

  if (code === 429 || /RESOURCE_EXHAUSTED|quota|429/i.test(msg)) {
    return 429;
  }

  if (/API key|GEMINI_API_KEY/i.test(msg)) {
    return 401;
  }

  return 500;
}

function sendError(
  res: express.Response,
  err: any,
  fallbackMessage: string
) {
  const status = errorStatus(err);
  let message = err?.message || fallbackMessage;

  if (status === 429) {
    message =
      'Превышена квота Gemini API. Подождите или выберите другую модель. ' +
      message;
  }

  res.status(status).json({ error: message });
}

type ThinkingHint = 'low' | 'medium' | 'high';

interface SamplingConfig {
  temperature?: number;
  topP?: number;
  topK?: number;
  frequencyPenalty?: number;
  presencePenalty?: number;
}

function resolveThinkingConfig(
  modelId: string,
  hint: ThinkingHint
): Record<string, any> | undefined {
  // gemini-3.1-flash-lite по умолчанию работает на MINIMAL.
  // MEDIUM/HIGH позволяют модели тщательнее планировать синтаксис.
  if (modelId === 'gemini-3.1-flash-lite') {
    const level =
      hint === 'high'
        ? ThinkingLevel.HIGH
        : hint === 'medium'
          ? ThinkingLevel.MEDIUM
          : ThinkingLevel.LOW;

    return { thinkingLevel: level };
  }

  if (modelId === 'gemini-4-hydrogen' && hint === 'low') {
    return { thinkingLevel: ThinkingLevel.LOW };
  }

  return undefined;
}

async function callModelOnce(params: {
  model: string;
  contents: string;
  systemInstruction: string;
  thinkingHint?: ThinkingHint;
  responseSchema?: any;
  sampling?: SamplingConfig;
}): Promise<string> {
  const {
    model,
    contents,
    systemInstruction,
    thinkingHint = 'medium',
    responseSchema,
    sampling,
  } = params;

  const thinkingConfig = resolveThinkingConfig(model, thinkingHint);

  const baseConfig: Record<string, any> = {
    systemInstruction,
  };

  if (responseSchema) {
    baseConfig.responseMimeType = 'application/json';
    baseConfig.responseSchema = responseSchema;
  }

  if (thinkingConfig) {
    baseConfig.thinkingConfig = thinkingConfig;
  }

  if (sampling) {
    if (typeof sampling.temperature === 'number') {
      baseConfig.temperature = sampling.temperature;
    }
    if (typeof sampling.topP === 'number') {
      baseConfig.topP = sampling.topP;
    }
    if (typeof sampling.topK === 'number') {
      baseConfig.topK = sampling.topK;
    }
    if (typeof sampling.frequencyPenalty === 'number') {
      baseConfig.frequencyPenalty = sampling.frequencyPenalty;
    }
    if (typeof sampling.presencePenalty === 'number') {
      baseConfig.presencePenalty = sampling.presencePenalty;
    }
  }

  try {
    const response = await ai.models.generateContent({
      model,
      contents,
      config: baseConfig,
    });

    return (response.text || '').trim();
  } catch (err: any) {
    const msg = String(err?.message || '');
    const isUnsupportedConfig =
      !isRateLimitError(err) &&
      ((baseConfig.thinkingConfig && /thinking/i.test(msg)) ||
        /frequencyPenalty|presencePenalty|frequency_penalty|presence_penalty|topK|unknown name|INVALID_ARGUMENT|not supported for this model/i.test(
          msg
        ));

    if (isUnsupportedConfig) {
      const retryConfig = { ...baseConfig };
      if (/thinking/i.test(msg)) {
        delete retryConfig.thinkingConfig;
      }
      delete retryConfig.frequencyPenalty;
      delete retryConfig.presencePenalty;
      delete retryConfig.topK;

      const response = await ai.models.generateContent({
        model,
        contents,
        config: retryConfig,
      });

      return (response.text || '').trim();
    }

    throw err;
  }
}

const modelCooldownUntil = new Map<string, number>();

function isRateLimitError(err: any): boolean {
  const msg = String(err?.message || '');
  const code = Number(err?.status || err?.code || 0);
  return code === 429 || /RESOURCE_EXHAUSTED|quota|rate[- ]limit|\b429\b/i.test(msg);
}

function parseQuotaCooldownMs(err: any): number {
  const msg = String(err?.message || '');
  const isDaily = /PerDay|GenerateRequestsPerDay|retry in \d+h/i.test(msg);
  const secMatch = msg.match(/"retryDelay"\s*:\s*"(\d+)s"/i);
  if (secMatch) {
    const ms = parseInt(secMatch[1], 10) * 1000;
    return isDaily ? Math.min(ms, 15 * 60 * 1000) : Math.min(Math.max(ms, 15_000), 90_000);
  }
  return isDaily ? 15 * 60 * 1000 : 60_000;
}

function markModelCooldown(modelId: string, err: any): void {
  if (!isRateLimitError(err)) return;
  const cooldownMs = parseQuotaCooldownMs(err);
  modelCooldownUntil.set(modelId, Date.now() + cooldownMs);
}

function isModelOnCooldown(modelId: string): boolean {
  const until = modelCooldownUntil.get(modelId);
  if (!until) return false;
  if (Date.now() >= until) {
    modelCooldownUntil.delete(modelId);
    return false;
  }
  return true;
}

function buildCandidateModelList(preferredModel: string): string[] {
  const normalized = normalizeModelId(preferredModel);
  const baseOrder = Array.from(
    new Set([normalized, 'gemini-3.1-flash-lite', 'gemini-4-hydrogen'])
  );
  const available = baseOrder.filter((m) => !isModelOnCooldown(m));
  // Если все модели на временном кулдауне, всё равно пробуем резервный порядок (начиная с high-quota Flash Lite)
  return available.length > 0 ? available : ['gemini-3.1-flash-lite', normalized];
}

function hashText(text: string): string {
  return crypto.createHash('sha256').update(text).digest('hex').slice(0, 24);
}

function setLru<K, V>(map: Map<K, V>, key: K, val: V, maxSize = 40): void {
  if (map.has(key)) map.delete(key);
  map.set(key, val);
  if (map.size > maxSize) {
    const oldest = map.keys().next().value;
    if (oldest !== undefined) map.delete(oldest);
  }
}

/**
 * Генерация с переходом на резервные модели и авто-охлаждением исчерпанных моделей.
 */
async function generateWithFallback(
  preferredModel: string,
  contents: string,
  systemInstruction: string,
  thinkingHint: ThinkingHint = 'medium',
  sampling?: SamplingConfig
): Promise<{
  text: string;
  usedModel: string;
  fallbackOccurred: boolean;
}> {
  if (!process.env.GEMINI_API_KEY) {
    throw new Error(
      'GEMINI_API_KEY не задан на сервере (проверьте .env или секреты проекта).'
    );
  }

  const normalized = normalizeModelId(preferredModel);
  const uniqueModels = buildCandidateModelList(normalized);

  let lastError: any = null;

  for (let i = 0; i < uniqueModels.length; i++) {
    const modelToTry = uniqueModels[i];

    try {
      const text = await callModelOnce({
        model: modelToTry,
        contents,
        systemInstruction,
        thinkingHint,
        sampling,
      });

      if (!text) {
        throw new Error(
          `Модель ${modelToTry} вернула пустой ответ (возможна блокировка по safety-фильтру).`
        );
      }

      return {
        text,
        usedModel: modelToTry,
        fallbackOccurred: modelToTry !== normalized,
      };
    } catch (err: any) {
      lastError = err;

      if (isRateLimitError(err)) {
        markModelCooldown(modelToTry, err);
        console.log(
          `[Gemini Quota] ${modelToTry} исчерпала лимит — автопереключение на резерв.`
        );
      } else {
        console.log(
          `[Gemini Fallback] ${modelToTry}: ${String(err?.message || err).slice(0, 180)}`
        );
      }

      if (
        /API key not valid|API_KEY_INVALID|GEMINI_API_KEY/i.test(
          String(err?.message || '')
        )
      ) {
        break;
      }
    }
  }

  throw lastError || new Error('Не удалось сгенерировать ответ через Gemini API');
}

/**
 * Безопасная генерация JSON.
 * При ошибке или некорректном JSON возвращает null.
 */
async function generateJsonSafe<T>(
  preferredModel: string,
  contents: string,
  systemInstruction: string,
  responseSchema: any,
  thinkingHint: ThinkingHint = 'low',
  sampling?: SamplingConfig
): Promise<T | null> {
  if (!process.env.GEMINI_API_KEY) {
    return null;
  }

  const models = buildCandidateModelList(preferredModel);

  for (const model of models) {
    try {
      const raw = await callModelOnce({
        model,
        contents,
        systemInstruction,
        thinkingHint,
        responseSchema,
        sampling,
      });

      if (!raw) {
        continue;
      }

      const cleaned = raw
        .replace(/^```(?:json)?\s*/i, '')
        .replace(/\s*```$/i, '')
        .trim();

      return JSON.parse(cleaned) as T;
    } catch (err: any) {
      if (isRateLimitError(err)) {
        markModelCooldown(model, err);
        console.log(`[Gemini JSON Quota] ${model} на паузе — переключение.`);
      } else {
        console.log(
          `[Gemini JSON Fallback] ${model}: ${String(err?.message || err).slice(0, 160)}`
        );
      }

      if (
        /API key not valid|API_KEY_INVALID|GEMINI_API_KEY/i.test(
          String(err?.message || '')
        )
      ) {
        break;
      }
    }
  }

  return null;
}

/**
 * Приводит список утраченных фактов к сопоставимому виду.
 *
 * Используется для проверки не только количества потерь,
 * но и конкретных элементов списка.
 */
function getCriticalFactKeys(diag: DiagnosticReport): Set<string> {
  const lostCritical = diag?.factsReport?.lostCritical;

  if (!Array.isArray(lostCritical)) {
    return new Set();
  }

  return new Set(
    lostCritical.map((fact: any) => {
      if (typeof fact === 'string') {
        return fact.trim().toLowerCase();
      }

      try {
        return JSON.stringify(fact);
      } catch {
        return String(fact);
      }
    })
  );
}

/**
 * Проверяет, что новый текст не потерял ни одного факта,
 * который уже присутствовал в предыдущем результате.
 *
 * Новые потери запрещены даже тогда, когда общее количество
 * утраченных фактов осталось прежним.
 */
function factsDoNotRegress(
  prevDiag: DiagnosticReport,
  nextDiag: DiagnosticReport
): boolean {
  const prevFacts = getCriticalFactKeys(prevDiag);
  const nextFacts = getCriticalFactKeys(nextDiag);

  for (const fact of nextFacts) {
    if (!prevFacts.has(fact)) {
      return false;
    }
  }

  return true;
}

/**
 * Единый критерий принятия кандидата.
 * Детекторные эвристики намеренно не участвуют в решении:
 * они остаются только диагностическими показателями.
 */
function acceptCandidateByDiagnostics(
  prevText: string,
  prevDiag: DiagnosticReport,
  nextText: string,
  nextDiag: DiagnosticReport
): boolean {
  const previous = prevText.trim();
  const next = nextText.trim();

  if (!next || next === previous) return false;
  if (!factsDoNotRegress(prevDiag, nextDiag)) return false;

  const prevLost = prevDiag.factsReport.lostCritical.length;
  const nextLost = nextDiag.factsReport.lostCritical.length;
  if (nextLost > prevLost) return false;

  const prevLeaks = prevDiag.factsReport.donorLeakTerms?.length ?? 0;
  const nextLeaks = nextDiag.factsReport.donorLeakTerms?.length ?? 0;
  if (nextLeaks > prevLeaks) return false;

  // Защита от случайного обрезания существенной части исходного результата.
  if (next.length < previous.length * 0.72) return false;

  const overallDelta = nextDiag.overallScore - prevDiag.overallScore;
  const dnaDelta = nextDiag.dnaMatchScore - prevDiag.dnaMatchScore;
  const humanDelta = nextDiag.humanScore - prevDiag.humanScore;

  const noMaterialRegression =
    overallDelta >= -1 && dnaDelta >= -2 && humanDelta >= -3;
  const meaningfulImprovement =
    overallDelta >= 0.5 || dnaDelta >= 1 || humanDelta >= 1;

  return noMaterialRegression && meaningfulImprovement;
}

function formatPropositionsInput(
  deSkeleton: DeSkeletonResult,
  rawFallbackText2: string,
  donorProfile: DonorDNAProfile
): string {
  const blocks = deSkeleton.paragraphs
    .map(
      (paragraph, index) =>
        `Блок ${index + 1} (${paragraph.theme}):\n${paragraph.propositions
          .map((prop) => `- ${prop}`)
          .join('\n')}`
    )
    .join('\n\n');

  const entities =
    deSkeleton.mandatoryEntities.length > 0
      ? `\nОБЯЗАТЕЛЬНЫЕ ИМЕНА, ДАТЫ И ТЕРМИНЫ (включить 100%): ${deSkeleton.mandatoryEntities.join(', ')}`
      : '';

  return `АТОМАРНЫЕ ТЕЗИСЫ И ФАКТЫ ДЛЯ НАПИСАНИЯ (ОЧИЩЕНО ОТ ИИ-КАРКАСА ТЕКСТА 2):
${blocks}
${entities}

=========================
ДЛЯ КОНТРОЛЯ ТОЧНОСТИ (ИСХОДНЫЙ ТЕКСТ 2 — проверяй факты, но НЕ копируй его синтаксис):
${rawFallbackText2}

=========================

ЗАДАНИЕ:
Собери из этих фактов цельный живой текст в абсолютном голосе, ритме (${donorProfile.avgSentPerPara} предл./абз., Burstiness ~${donorProfile.burstiness}) и синтаксисе Автора-Донора.

Выведи ТОЛЬКО готовый текст без заголовков, комментариев и кавычек.`;
}

function inferGenreFromInput(text: string, hint: string): 'academic' | 'essay' | 'draft' {
  const combined = `${hint}\n${text.slice(0, 1200)}`.toLowerCase();
  if (/диссертац|монограф|научн|методолог|гипотез|таблиц|выборк|эксперимент|исследован|академич|статья/i.test(combined)) {
    return 'academic';
  }
  if (/эссе|публицист|очерк|живой|истори|философ|размышлен|колонка/i.test(combined)) {
    return 'essay';
  }
  return 'draft';
}

/**
 * Скользящий поабзацный перенос длинных текстов (2200+ знаков, >= 4 смысловых блока):
 * переносит блоками по 2 абзаца с передачей последних 2 предложений предыдущего блока
 * для сохранения ритмической и смысловой преемственности без «усталости» на длинном хвосте.
 */

 */
async function generateChunkedStyleTransfer(params: {
  deSkeleton: DeSkeletonResult;
  donorProfile: DonorDNAProfile;
  systemInstruction: string;
  model: string;
}): Promise<{ text: string; usedModel: string; fallbackOccurred: boolean }> {
  const { deSkeleton, donorProfile, systemInstruction, model } = params;
  const chunks: Array<DeSkeletonResult['paragraphs']> = [];
  for (let i = 0; i < deSkeleton.paragraphs.length; i += 2) {
    chunks.push(deSkeleton.paragraphs.slice(i, i + 2));
  }

  const generatedParts: string[] = [];
  let usedModel = model;
  let fallbackOccurred = false;

  for (let idx = 0; idx < chunks.length; idx++) {
    const chunkParas = chunks[idx];
    const blocksText = chunkParas
      .map(
        (p, pIdx) =>
          `Блок ${idx * 2 + pIdx + 1} (${p.theme}):\n${p.propositions
            .map((prop) => `- ${prop}`)
            .join('\n')}`
      )
      .join('\n\n');

    const bridgeText =
      idx > 0 && generatedParts.length > 0
        ? `\n\nПРЕДЫДУЩИЙ ФРАГМЕНТ (только для точного ритмического стыка — НЕ повторяй эти фразы):\n«${extractLastSentences(generatedParts[generatedParts.length - 1], 2)}»`
        : '';

    const chunkPrompt = `ЧАСТЬ ${idx + 1} ИЗ ${chunks.length} (ПОСЛЕДОВАТЕЛЬНЫЙ ПЕРЕНОС ПО ДОНОРУ):
${blocksText}

КЛЮЧЕВЫЕ ИМЕНА И ТЕРМИНЫ ТЕКСТА 2 (сохранить те, что относятся к этим блокам): ${deSkeleton.mandatoryEntities.join(', ') || 'по смыслу блоков'}${bridgeText}

ОРИЕНТИР РИТМА ДОНОРА: ${donorProfile.avgSentPerPara} предл./абз., средняя длина фразы ${donorProfile.avgLen} слов, тип финала абзаца: ${donorProfile.paragraphClosingStyle}.
Напиши ТОЛЬКО эту часть (${chunkParas.length} абз.) чистым текстом без заголовков и комментариев.`;

    const res = await generateWithFallback(
      usedModel,
      chunkPrompt,
      systemInstruction,
      idx === 0 ? 'high' : 'medium',
      { temperature: 0.96, topP: 0.95, frequencyPenalty: 0.3, presencePenalty: 0.2 }
    );

    usedModel = res.usedModel;
    fallbackOccurred = fallbackOccurred || res.fallbackOccurred;
    generatedParts.push(deterministicPolish(res.text, donorProfile));
  }

  return {
    text: deterministicPolish(generatedParts.join('\n\n'), donorProfile),
    usedModel,
    fallbackOccurred,
  };
}

 *
 * Можно передать уже рассчитанную диагностику, чтобы не выполнять
 * повторный анализ того же самого текста.
 */
async function runGenericConvergenceLoop(params: {
  initialText: string;
  sourceText: string;
  model: string;
  maxCycles: number;
  initialDiagnostic?: DiagnosticReport;
  shields?: DetectorShields;
}): Promise<GenericConvergenceResult> {
  const {
    initialText,
    sourceText,
    model,
    maxCycles,
    initialDiagnostic,
    shields,
  } = params;

  let currentText = deterministicPolish(initialText);

  let currentDiag =
    initialDiagnostic ||
    diagnoseTextDefects(currentText, undefined, sourceText);

  let currentDetector = evaluateDetectorFeedback(currentText, shields);

  const diagBefore = currentDiag;
  const detectorAiBefore = currentDetector.compositeAiPct;

  let currentModel = model;
  let fallbackOccurred = false;

  let passes = 1;
  let stalled = 0;

  for (let cycle = 1; cycle <= maxCycles; cycle++) {
    // Если качество текста уже достаточно высокое, дополнительные запросы не нужны.
    if (
      !currentDiag.needsRefinement &&
      currentDiag.overallScore >= 94
    ) {
      break;
    }

    // Две неудачные попытки подряд означают, что дальнейшие запросы
    // с высокой вероятностью не дадут полезного улучшения.
    if (stalled >= 2) {
      break;
    }

    try {
      const hasGlobalIssue = currentDiag.issues.some(
        (issue) =>
          issue.type === 'burstiness' ||
          issue.type === 'donorLengthDrift' ||
          issue.type === 'factLoss' ||
          issue.type === 'ttr' ||
          issue.type === 'donorMarkers' ||
          issue.type === 'nominalizationOverload' ||
          issue.type === 'negationShift'
      );

      const combinedLocalIds = Array.from(
        new Set(currentDiag.issues.flatMap((i) => i.sentenceIndexes))
      ).filter((id) => id > 0);

      const hasOnlyLocal =
        combinedLocalIds.length > 0 &&
        !hasGlobalIssue &&
        currentDiag.issues.every((issue) => issue.sentenceIndexes.length > 0);

      let candidateRaw = '';

      // Хирургическая правка используется только при локальных дефектах диагностики.
      if (hasOnlyLocal && stalled === 0) {
        const patchPrompt = buildSurgicalPatchPrompt(
          currentText,
          currentDiag,
          undefined,
          null,
          []
        );

        const patchRes = await generateJsonSafe<{
          patches: Array<{
            sentenceId: number;
            rewritten: string;
          }>;
        }>(
          'gemini-3.1-flash-lite',
          patchPrompt.contents,
          patchPrompt.systemInstruction,
          SURGICAL_PATCH_RESPONSE_SCHEMA,
          'medium',
          { temperature: 0.4, topP: 0.9 }
        );

        if (patchRes?.patches?.length) {
          candidateRaw = applySentencePatches(
            currentText,
            patchRes.patches
          );
        }
      }

      // Если локальная правка не дала результата или проблема глобальная,
      // используем общий refinement по содержательной диагностике.
      if (!candidateRaw) {
        const refine = buildRefinementPrompt(
          currentText,
          currentDiag,
          undefined,
          null,
          [],
          sourceText
        );

        const passRes = await generateWithFallback(
          currentModel,
          refine.contents,
          refine.systemInstruction,
          'high',
          { temperature: 0.86, topP: 0.94, frequencyPenalty: 0.25 }
        );

        candidateRaw = passRes.text;

        currentModel = passRes.usedModel;
        fallbackOccurred =
          fallbackOccurred || passRes.fallbackOccurred;
      }

      const candidate = deterministicPolish(candidateRaw);

      const candidateDiag = diagnoseTextDefects(
        candidate,
        undefined,
        sourceText
      );
      const candidateDetector = evaluateDetectorFeedback(candidate, shields);

      if (
        acceptCandidateByDiagnostics(
          currentText,
          currentDiag,
          candidate,
          candidateDiag
        )
      ) {
        currentText = candidate;
        currentDiag = candidateDiag;
        currentDetector = candidateDetector;

        passes++;
        stalled = 0;
      } else {
        stalled++;
      }
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      console.log(
        `[Convergence Loop cycle ${cycle}] stopped cleanly: ${msg.slice(0, 180)}`
      );

      break;
    }
  }

  return {
    finalText: currentText,
    diagBefore,
    diagAfter: currentDiag,
    detectorAiBefore,
    detectorAiAfter: currentDetector.compositeAiPct,
    passes,
    usedModel: currentModel,
    fallbackOccurred,
  };
}

app.post('/api/style-dna-transfer', async (req, res) => {
  try {
    const {
      text1,
      text2,
      flags = {},
      shields = {},
      enable20CycleCheck,
      model = 'gemini-3.1-flash-lite',
    } = req.body;

    if (
      typeof text1 !== 'string' ||
      typeof text2 !== 'string' ||
      !text1.trim() ||
      !text2.trim()
    ) {
      return res.status(400).json({
        error: 'Текст 1 и Текст 2 обязательны',
      });
    }

    const runDeepConvergence: boolean =
      enable20CycleCheck ?? !!flags.twentyCycles;

    const normalizedModel = normalizeModelId(model);

    const donorProfile = extractDonorDNAProfile(text1);
    const donorDetectorBaseline = computeDonorDetectorBaseline(text1, shields);

    /* -------------------------------------------------------------- */
    /* ЭТАП 1. ЧЕРТЁЖ ДОНОРА И ДЕ-СКЕЛЕТИЗАЦИЯ (С КЭШЕМ + FLASH LITE)*/
    /* -------------------------------------------------------------- */

    let styleBlueprint: DonorStyleBlueprint | null = null;
    let deSkeleton: DeSkeletonResult | null = null;

    if (runDeepConvergence) {
      const t1Hash = hashText(text1.trim());
      const t2Hash = hashText(text2.trim());

      const cachedBp = blueprintCache.get(t1Hash) || null;
      const cachedDs = deSkeletonCache.get(t2Hash) || null;

      // Вспомогательные JSON-этапы всегда пускаем через высококвотную Flash Lite,
      // чтобы не сжигать дневной лимит gemini-4-hydrogen (20 req/day на Free Tier).
      const helperModel = 'gemini-3.1-flash-lite';

      const [bpRes, dsRes] = await Promise.all([
        cachedBp
          ? Promise.resolve(cachedBp)
          : (() => {
              const bpPrompt = buildDonorBlueprintPrompt(text1);
              return generateJsonSafe<DonorStyleBlueprint>(
                helperModel,
                bpPrompt.contents,
                bpPrompt.systemInstruction,
                BLUEPRINT_RESPONSE_SCHEMA,
                'medium',
                { temperature: 0.35, topP: 0.9 }
              );
            })(),

        cachedDs
          ? Promise.resolve(cachedDs)
          : text2.trim().length >= 180
          ? (() => {
              const dsPrompt = buildDeSkeletonPrompt(text2);
              return generateJsonSafe<DeSkeletonResult>(
                helperModel,
                dsPrompt.contents,
                dsPrompt.systemInstruction,
                PROPOSITIONS_RESPONSE_SCHEMA,
                'medium',
                { temperature: 0.3, topP: 0.9 }
              );
            })()
          : Promise.resolve(null),
      ]);

      if (bpRes) {
        styleBlueprint = bpRes;
        if (!cachedBp) setLru(blueprintCache, t1Hash, bpRes);
      }

      if (dsRes && dsRes.paragraphs?.length > 0) {
        deSkeleton = dsRes;
        if (!cachedDs) setLru(deSkeletonCache, t2Hash, dsRes);
      }
    }

    /* -------------------------------------------------------------- */
    /* СИСТЕМНЫЙ РЕГЛАМЕНТ И АНТИ-УТЕЧКА ДОНОРА                       */
    /* -------------------------------------------------------------- */

    const forbiddenDonorTerms = extractDonorForbiddenTerms(text1, text2);

    const promptSections = buildEnginePromptSections({
      flags,
      shields,
      enable20CycleCheck: runDeepConvergence,
      donorProfile,
      styleBlueprint,
      forbiddenDonorTerms,
      mode: 'style-dna',
    });

    const systemInstruction = promptSections.join('\n\n');

    const directContents = `ТЕКСТ 1 (ОБРАЗЕЦ СТИЛЯ — ДОНОР, КОПИРУЙ ТОЛЬКО СИНТАКСИС И ГОЛОС):
${text1}

=========================

ТЕКСТ 2 (МАТЕРИАЛ ДЛЯ ПЕРЕРАБОТКИ — СОХРАНИ 100% ФАКТОВ):
${text2}

=========================

Выведи ТОЛЬКО готовый результирующий текст без вступительных реплик, без кавычек и без послесловий.`;

    /* -------------------------------------------------------------- */
    /* ЭТАП 2. ГЕНЕРАЦИЯ КАНДИДАТОВ (MULTI-TEMP SAMPLING)             */
    /* -------------------------------------------------------------- */

    let bestText = '';
    let bestDiag: DiagnosticReport | null = null;
    let bestDet: DetectorFeedbackReport | null = null;
    let chunkedTransfer = false;

    let usedModel = normalizedModel;
    let fallbackOccurred = false;

    const preferHighQuotaDual =
      !isModelOnCooldown('gemini-3.1-flash-lite') &&
      (normalizedModel === 'gemini-3.1-flash-lite' || isModelOnCooldown(normalizedModel));

    const shouldUseChunkedCandidate = Boolean(
      runDeepConvergence &&
        deSkeleton &&
        deSkeleton.paragraphs.length >= 4 &&
        text2.trim().length >= 2200
    );

    if (runDeepConvergence && deSkeleton && preferHighQuotaDual) {
      const propContents = formatPropositionsInput(
        deSkeleton,
        text2,
        donorProfile
      );

      const candidatePromises: Array<
        Promise<{
          text: string;
          usedModel: string;
          fallbackOccurred: boolean;
          isChunked?: boolean;
        }>
      > = [
        // Кандидат A: де-скелет + сбалансированная живость (T=0.92)
        generateWithFallback(
          normalizedModel,
          propContents,
          systemInstruction,
          'high',
          { temperature: 0.92, topP: 0.95, frequencyPenalty: 0.25, presencePenalty: 0.15 }
        ),
        // Кандидат B: прямой перенос + высокая синтаксическая вариативность (T=1.12)
        generateWithFallback(
          normalizedModel,
          directContents,
          systemInstruction,
          'medium',
          { temperature: 1.12, topP: 0.96, frequencyPenalty: 0.35, presencePenalty: 0.25 }
        ),
        // Кандидат C: на длинных текстах — скользящий поабзацный перенос, иначе high-temp (T=1.24)
        shouldUseChunkedCandidate
          ? generateChunkedStyleTransfer({
              deSkeleton,
              donorProfile,
              systemInstruction,
              model: 'gemini-3.1-flash-lite',
            }).then((r) => ({ ...r, isChunked: true }))
          : generateWithFallback(
              'gemini-3.1-flash-lite',
              propContents,
              systemInstruction,
              'low',
              { temperature: 1.24, topP: 0.97, frequencyPenalty: 0.4, presencePenalty: 0.3 }
            ),
      ];

      const settledResults = await Promise.allSettled(candidatePromises);

      const candidates: Array<{
        text: string;
        diag: DiagnosticReport;
        det: DetectorFeedbackReport;
        model: string;
        fallback: boolean;
        isChunked: boolean;
      }> = [];

      for (const result of settledResults) {
        if (result.status !== 'fulfilled' || !result.value.text) {
          continue;
        }

        const polished = deterministicPolish(
          result.value.text,
          donorProfile
        );

        const diag = diagnoseTextDefects(
          polished,
          donorProfile,
          text2
        );
        const det = evaluateDetectorFeedback(
          polished,
          shields,
          donorDetectorBaseline
        );

        candidates.push({
          text: polished,
          diag,
          det,
          model: result.value.usedModel,
          fallback: result.value.fallbackOccurred,
          isChunked: Boolean(result.value.isChunked),
        });
      }

      if (candidates.length === 0) {
        const firstRejected = settledResults.find(
          (r): r is PromiseRejectedResult => r.status === 'rejected'
        );
        throw (
          firstRejected?.reason ||
          new Error('Все параллельные попытки генерации завершились без результата.')
        );
      }

      // Сначала сравниваем сохранность фактов и отсутствие утечки темы донора,
      // затем качество переноса по содержательной диагностике и совпадению Style DNA.
      candidates.sort((a, b) => {
        const lostDiff =
          a.diag.factsReport.lostCritical.length -
          b.diag.factsReport.lostCritical.length;

        if (lostDiff !== 0) {
          return lostDiff;
        }

        const leakDiff =
          (a.diag.factsReport.donorLeakTerms?.length ?? 0) -
          (b.diag.factsReport.donorLeakTerms?.length ?? 0);

        if (leakDiff !== 0) {
          return leakDiff;
        }

        if (a.diag.overallScore !== b.diag.overallScore) {
          return b.diag.overallScore - a.diag.overallScore;
        }

        if (a.diag.dnaMatchScore !== b.diag.dnaMatchScore) {
          return b.diag.dnaMatchScore - a.diag.dnaMatchScore;
        }

        return b.diag.humanScore - a.diag.humanScore;
      });

      bestText = candidates[0].text;
      bestDiag = candidates[0].diag;
      bestDet = candidates[0].det;
      usedModel = candidates[0].model;
      fallbackOccurred = candidates[0].fallback;
      chunkedTransfer = candidates[0].isChunked;
    } else {
      // Для малолимитной Hydrogen (20 req/day) делаем 1 точный синтез, опираясь на де-скелет/чертёж от Flash Lite
      const primaryInput =
        deSkeleton
          ? formatPropositionsInput(deSkeleton, text2, donorProfile)
          : directContents;

      const pass1 = await generateWithFallback(
        normalizedModel,
        primaryInput,
        systemInstruction,
        'high',
        { temperature: 0.98, topP: 0.95, frequencyPenalty: 0.25, presencePenalty: 0.15 }
      );

      bestText = deterministicPolish(
        pass1.text,
        donorProfile
      );

      bestDiag = diagnoseTextDefects(
        bestText,
        donorProfile,
        text2
      );
      bestDet = evaluateDetectorFeedback(
        bestText,
        shields,
        donorDetectorBaseline
      );

      usedModel = pass1.usedModel;
      fallbackOccurred =
        pass1.fallbackOccurred || pass1.usedModel !== normalizedModel;
    }

    if (!bestDiag || !bestDet) {
      throw new Error('Не удалось сформировать диагностический отчёт.');
    }

    /* -------------------------------------------------------------- */
    /* ЭТАП 3. ПРОВЕРКА ПЕРЕНОСА И СХОДИМОСТЬ ПО STYLE DNA             */
    /* -------------------------------------------------------------- */

    const diagBefore = bestDiag;
    const detectorBefore = bestDet;

    const factsLostBefore =
      diagBefore.factsReport.lostCritical.length;

    let passes = 1;
    let cyclesRun = 0;

    let criticScore: number | null = null;
    let criticRuns = 0;
    let lastCriticPass = 0;

    let pendingCriticDirectives: string[] = [];
    let pendingCriticIds: number[] = [];

    let stopReason = '100% допуск с первого синтеза';

    const MAX_CONVERGENCE_CYCLES = 6;

    let stalledCycles = 0;



    if (runDeepConvergence) {
      for (
        let cycle = 1;
        cycle <= MAX_CONVERGENCE_CYCLES;
        cycle++
      ) {
        /* ---------------------------------------------------------- */
        /* НЕЗАВИСИМЫЙ LLM-КРИТИК (до 2 проходов: старт >= 76 и финал) */
        /* ---------------------------------------------------------- */

        const canInvokeCritic =
          bestDiag.factsReport.lostCritical.length === 0 &&
          (bestDiag.factsReport.donorLeakTerms?.length ?? 0) === 0 &&
          bestDiag.overallScore >= 76 &&
          criticRuns < 2 &&
          (criticRuns === 0 || passes > lastCriticPass);

        if (canInvokeCritic) {
          criticRuns++;
          lastCriticPass = passes;

          try {
            const criticPrompt = buildStyleCriticPrompt(
              text1,
              text2,
              bestText,
              donorProfile,
              styleBlueprint
            );

            const criticRes =
              await generateJsonSafe<CriticResult>(
                'gemini-3.1-flash-lite',
                criticPrompt.contents,
                criticPrompt.systemInstruction,
                CRITIC_RESPONSE_SCHEMA,
                'medium',
                { temperature: 0.25, topP: 0.9 }
              );

            if (!criticRes) {
              stopReason =
                'Пройдены все алгоритмические проверки качества';
              break;
            }

            const scores = [
              criticRes.styleFidelityScore,
              criticRes.humanNaturalnessScore,
              criticRes.factCompletenessScore,
            ];

            // Некорректные оценки не должны влиять на решение.
            const validScores = scores.every(
              (score) =>
                typeof score === 'number' &&
                Number.isFinite(score) &&
                score >= 0 &&
                score <= 100
            );

            if (!validScores) {
              stopReason =
                'Критик вернул некорректные оценки; сохранён лучший результат';
              break;
            }

            criticScore = Math.round(
              scores.reduce((sum, score) => sum + score, 0) /
                scores.length
            );

            if (
              criticRes.passed &&
              criticRes.styleFidelityScore >= 88 &&
              criticRes.factCompletenessScore >= 90
            ) {
              stopReason =
                `Пройдены проверки сохранности фактов и Style DNA; оценка критика ${criticScore}/100`;
              break;
            }

            pendingCriticDirectives = Array.isArray(
              criticRes.actionableDirectives
            )
              ? criticRes.actionableDirectives
                  .filter(
                    (directive) =>
                      typeof directive === 'string' &&
                      directive.trim().length > 0
                  )
                  .slice(0, 5)
              : [];

            pendingCriticIds = Array.isArray(
              criticRes.suspiciousSentenceIds
            )
              ? criticRes.suspiciousSentenceIds
                  .filter(
                    (id) =>
                      Number.isInteger(id) &&
                      id >= 0
                  )
                  .slice(0, 6)
              : [];

            // Если критик не прошёл проверку, но не указал,
            // что именно нужно исправить, новый запрос бессмысленен.
            if (
              !criticRes.passed &&
              pendingCriticDirectives.length === 0 &&
              pendingCriticIds.length === 0
            ) {
              stopReason =
                'Критик не предоставил конкретных замечаний; сохранён лучший результат';
              break;
            }
          } catch (criticError) {
            const msg = criticError instanceof Error ? criticError.message : String(criticError);
            console.log(
              `[Style Critic] skipped: ${msg.slice(0, 180)}`
            );

            stopReason =
              'Пройдены алгоритмические проверки; критик недоступен';
            break;
          }
        } else if (
          !bestDiag.needsRefinement &&
          bestDiag.overallScore >= 94 &&
          criticRuns >= 1
        ) {
          stopReason =
            `Достигнут целевой уровень качества (${bestDiag.overallScore}/100)`;
          break;
        }

        if (stalledCycles >= 2) {
          stopReason =
            'Достигнут устойчивый максимум (дальнейшие правки не улучшают метрику)';
          break;
        }

        cyclesRun++;

        try {
          /* -------------------------------------------------------- */
          /* ОПРЕДЕЛЕНИЕ ГЛОБАЛЬНЫХ ДЕФЕКТОВ                          */
          /* -------------------------------------------------------- */

          const hasGlobalIssue = bestDiag.issues.some(
            (issue) =>
              issue.type === 'burstiness' ||
              issue.type === 'donorLengthDrift' ||
              issue.type === 'factLoss' ||
              issue.type === 'ttr' ||
              issue.type === 'donorMarkers' ||
              issue.type === 'nominalizationOverload' ||
              issue.type === 'negationShift'
          );

          const mergedCriticDirectives = pendingCriticDirectives.slice(0, 5);

          const mergedLocalIds = Array.from(
            new Set([
              ...pendingCriticIds,
              ...bestDiag.issues.flatMap((issue) => issue.sentenceIndexes),
            ])
          )
            .filter((id) => Number.isInteger(id) && id > 0)
            .slice(0, 6);

          const hasLocalIds =
            bestDiag.issues.some(
              (issue) => issue.sentenceIndexes.length > 0
            ) ||
            mergedLocalIds.length > 0;

          let candidateRaw = '';

          /* -------------------------------------------------------- */
          /* ХИРУРГИЧЕСКАЯ ПРАВКА                                     */
          /* -------------------------------------------------------- */

          if (
            !hasGlobalIssue &&
            hasLocalIds &&
            stalledCycles === 0
          ) {
            const syntheticDiag: DiagnosticReport =
              mergedLocalIds.length > 0 &&
              bestDiag.issues.length === 0
                ? {
                    ...bestDiag,
                    issues: [
                      {
                        type: 'cliche',
                        severity: 'major',
                        sentenceIndexes: mergedLocalIds,
                        message:
                          `Предложения №${mergedLocalIds.join(', №')} ` +
                          'требуют проверки по содержательной диагностике и соответствию стилю донора.',
                      },
                    ],
                  }
                : bestDiag;

            const patchPrompt = buildSurgicalPatchPrompt(
              bestText,
              syntheticDiag,
              donorProfile,
              styleBlueprint,
              mergedCriticDirectives
            );

            const patchRes = await generateJsonSafe<{
              patches: Array<{
                sentenceId: number;
                rewritten: string;
              }>;
            }>(
              'gemini-3.1-flash-lite',
              patchPrompt.contents,
              patchPrompt.systemInstruction,
              SURGICAL_PATCH_RESPONSE_SCHEMA,
              'medium',
              { temperature: 0.42, topP: 0.9 }
            );

            if (patchRes?.patches?.length) {
              candidateRaw = applySentencePatches(
                bestText,
                patchRes.patches
              );
            }
          }

          /* -------------------------------------------------------- */
          /* ОБЩИЙ REFINEMENT                                         */
          /* -------------------------------------------------------- */

          if (!candidateRaw) {
            const refinePrompt = buildRefinementPrompt(
              bestText,
              bestDiag,
              donorProfile,
              styleBlueprint,
              mergedCriticDirectives,
              text2
            );

            const passN = await generateWithFallback(
              usedModel,
              refinePrompt.contents,
              refinePrompt.systemInstruction,
              'high',
              { temperature: 0.85, topP: 0.94, frequencyPenalty: 0.25 }
            );

            candidateRaw = passN.text;

            // Обновляем информацию о модели, если сработал fallback.
            usedModel = passN.usedModel;

            fallbackOccurred =
              fallbackOccurred || passN.fallbackOccurred;
          }

          const candidate = deterministicPolish(
            candidateRaw,
            donorProfile
          );

          const candidateDiag = diagnoseTextDefects(
            candidate,
            donorProfile,
            text2
          );
          const candidateDet = evaluateDetectorFeedback(
            candidate,
            shields,
            donorDetectorBaseline
          );

          /* -------------------------------------------------------- */
          /* ПРОВЕРКА КАНДИДАТА ПО ФАКТАМ И КАЧЕСТВУ STYLE DNA       */
          /* -------------------------------------------------------- */

          if (
            acceptCandidateByDiagnostics(
              bestText,
              bestDiag,
              candidate,
              candidateDiag
            )
          ) {
            bestText = candidate;
            bestDiag = candidateDiag;
            bestDet = candidateDet;

            passes++;
            stalledCycles = 0;

            pendingCriticDirectives = [];
            pendingCriticIds = [];
          } else {
            stalledCycles++;
          }
        } catch (cycleError) {
          const msg = cycleError instanceof Error ? cycleError.message : String(cycleError);
          console.log(
            `[Convergence Cycle ${cycle}] gracefully stopped: ${msg.slice(0, 180)}`
          );

          stopReason =
            'Завершено досрочно (ограничение API на позднем цикле), сохранён лучший результат';

          break;
        }
      }

function isPrivateOrBlockedIp(ip: string): boolean {
  const cleanIp = ip.toLowerCase();
  if (cleanIp === '::1' || cleanIp === '::' || cleanIp.startsWith('fe80:') || cleanIp.startsWith('fc') || cleanIp.startsWith('fd')) {
    return true;
  }
  const v4 = cleanIp.startsWith('::ffff:') ? cleanIp.slice(7) : cleanIp;
  if (net.isIPv4(v4)) {
    const parts = v4.split('.').map(Number);
    const [a, b] = parts;
    if (a === 0 || a === 10 || a === 127) return true;
    if (a === 169 && b === 254) return true;
    if (a === 172 && b >= 16 && b <= 31) return true;
    if (a === 192 && b === 168) return true;
    if (a === 100 && b >= 64 && b <= 127) return true;
  }
  return false;
}

  try {

  }

  if (currentUrl.protocol !== 'http:' && currentUrl.protocol !== 'https:') {

  }

  const MAX_REDIRECTS = 3;
  const MAX_BYTES = 5 * 1024 * 1024;

      throw new Error('Доступ к локальным или внутренним адресам запрещён');
    }

    if (net.isIP(hostname)) {
      if (isPrivateOrBlockedIp(hostname)) {
        throw new Error('Заблокирован приватный или loopback IP-адрес');
      }
    } else {
      const resolved = await dns.promises.lookup(hostname, { all: true });
      if (!resolved || resolved.length === 0) {
        throw new Error(`Не удалось разрешить домен: ${hostname}`);
      }
      for (const entry of resolved) {

        },
      });

      if (response.status >= 300 && response.status < 400) {
        const loc = response.headers.get('location');
        if (!loc) {
          throw new Error(`Сервер вернул редирект ${response.status} без заголовка Location`);
        }
        currentUrl = new URL(loc, currentUrl);

      }

      const contentLength = Number(response.headers.get('content-length') || 0);

      if (!reader) {

      } else {

        let totalBytes = 0;

        try {

            const { done, value } = await reader.read();
            if (done) break;
            if (!value) continue;

            totalBytes += value.byteLength;
            if (totalBytes > MAX_BYTES) {
              await reader.cancel();
              throw new Error('Размер ответа превышает лимит 5 МБ');
            }

            chunks.push(Buffer.from(value));
          }
        } finally {
          reader.releaseLock();
        }

        buf = Buffer.concat(chunks, totalBytes);
      }
    } finally {
      clearTimeout(timer);
    }

    if (!buf) {

    }

    const headersObj: Record<string, string> = {};
    response.headers.forEach((v, k) => {
      headersObj[k.toLowerCase()] = v;
    });

    return {

      html: buf.toString('utf8'),
      headers: headersObj,
      status: response.status,

app.post('/api/mgt-eval/attack', (req, res) => {
  try {
    const {
      text = '',
      attackType = 'homo',
      pct = 0.25,
      seed = 42,
      detectorKeys,
      threshold = 0.5,
    } = req.body || {};

    const cleanText =
      typeof text === 'string'
        ? text.trim()
        : '';

    if (cleanText.length < 15) {
      return res.status(400).json({
        error: 'Введите минимум 15 символов для симуляции атаки и расчёта ASR.',
      });
    }

    const validAttack = MGT_ATTACK_CATALOG.some(
      (attack) => attack.type === attackType
    )
      ? (attackType as MgtAttackType)
      : 'homo';

    const ratio =
      typeof pct === 'number' && Number.isFinite(pct)
        ? Math.max(0.05, Math.min(0.8, pct))
        : 0.25;

    const thr =
      typeof threshold === 'number' && Number.isFinite(threshold)
        ? Math.max(0, Math.min(1, threshold))
        : 0.5;

    const validKeys = Array.isArray(detectorKeys)
      ? detectorKeys.filter(
          (key: unknown): key is string =>
            typeof key === 'string'
        )
      : undefined;

    const numericSeed = Number(seed);
    const safeSeed =
      Number.isFinite(numericSeed)
        ? numericSeed
        : 42;

    const attackedText = applySingleMgtAttack(
      cleanText,
      validAttack,
      ratio,
      safeSeed
    );

    const qualityBefore = evaluateMgtQuality(cleanText);
    const qualityAfter = evaluateMgtQuality(
      attackedText,
      cleanText
    );

    const beforeReport = runMgtEvalDetection(cleanText, {
      detectorKeys: validKeys,
      threshold: thr,
    });

    const afterReport = runMgtEvalDetection(attackedText, {
      detectorKeys: validKeys,
      threshold: thr,
    });

    let previouslyFlagged = 0;
    let evadedCount = 0;

    const comparison = beforeReport.results.map(
      (beforeResult, index) => {
        const afterResult = afterReport.results[index];

        const beforePrediction = beforeResult.prediction;
        const afterPrediction =
          afterResult?.prediction || beforePrediction;

        const flippedToHuman =
          beforePrediction === 'ai' &&
          afterPrediction === 'human';

        if (beforePrediction === 'ai') {
          previouslyFlagged += 1;
        }

        if (flippedToHuman) {
          evadedCount += 1;
        }

        const afterProbability =
          afterResult?.aiProbability ??
          beforeResult.aiProbability;

        return {
          key: beforeResult.key,
          name: beforeResult.name,
          type: beforeResult.type,

          beforeAiProb: beforeResult.aiProbability,
          afterAiProb: afterProbability,

          deltaAiProb: Number(
            (afterProbability - beforeResult.aiProbability).toFixed(4)
          ),

          beforePrediction,
          afterPrediction,
          flippedToHuman,
        };
      }
    );

    const asr =
      previouslyFlagged > 0
        ? Number(
            (evadedCount / previouslyFlagged).toFixed(4)
          )
        : 0;

    const avgAiDrop = Number(
      (
        beforeReport.overallAiProbability -
        afterReport.overallAiProbability
      ).toFixed(4)
    );

    mgtEvalStats.totalAttacks += 1;

    res.json({
      attackType: validAttack,
      pct: ratio,

      originalText: cleanText,
      attackedText,

      qualityBefore,
      qualityAfter,

      beforeOverallAi: beforeReport.overallAiProbability,
      afterOverallAi: afterReport.overallAiProbability,

      avgAiDrop,
      previouslyFlagged,
      evadedCount,
      asr,

      comparison,
    });
  } catch (err: any) {
    console.error('MGT-Eval attack error:', err);
    res.status(400).json({

    });
  }

const mgtBenchStats = {
  totalDetections: 0,
  totalAttributions: 0,
  totalBenchmarks: 0,
};

function parseMgtBenchSamples(
  rawInput: unknown
): MgtBenchSample[] {
  if (Array.isArray(rawInput)) {
    const out: MgtBenchSample[] = [];

    for (let i = 0; i < rawInput.length; i++) {
      const item: any = rawInput[i];

      if (
        item &&
        typeof item.text === 'string' &&
        item.text.trim()
      ) {
        const label = Number(item.label) > 0.5 ? 1 : 0;

        out.push({
          dataset:
            typeof item.dataset === 'string'
              ? item.dataset
              : 'Custom',

          llm:
            typeof item.llm === 'string'
              ? item.llm
              : label === 1
                ? 'ChatGPT'
                : 'Human',

          label,
          text: item.text.trim(),
        });
      }
    }

    return out;
  }

  if (
    typeof rawInput === 'string' &&
    rawInput.trim()
  ) {
    const lines = rawInput
      .split(/\r?\n/)
      .map((line) => line.trim())
      .filter(Boolean);

    const out: MgtBenchSample[] = [];

    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];

      if (line.startsWith('{')) {
        try {
          const obj = JSON.parse(line);

          if (
            obj &&
            typeof obj.text === 'string' &&
            obj.text.trim()
          ) {
            const label = Number(obj.label) > 0.5 ? 1 : 0;

            out.push({
              dataset:
                typeof obj.dataset === 'string'
                  ? obj.dataset
                  : 'Custom',

              llm:
                typeof obj.llm === 'string'
                  ? obj.llm
                  : label === 1
                    ? 'ChatGPT'
                    : 'Human',

              label,
              text: obj.text.trim(),
            });
          }

          continue;
        } catch {
          // При ошибке JSON пробуем разобрать строку как CSV/TSV.
        }
      }

      if (
        i === 0 &&
        /^label[,;\t]/i.test(line)
      ) {
        continue;
      }

      const tabParts = line.split('\t');

      if (
        tabParts.length >= 2 &&
        /^[01]$/.test(tabParts[0].trim())
      ) {
        const label = Number(tabParts[0].trim());

        out.push({
          dataset: 'TSV',
          llm: label === 1 ? 'ChatGPT' : 'Human',
          label,
          text: tabParts.slice(1).join('\t').trim(),
        });

        continue;
      }

      const csvMatch = line.match(
        /^([01])\s*[,;]\s*"?(.*?)"?$/
      );

      if (csvMatch && csvMatch[2].trim()) {
        const label = Number(csvMatch[1]);

        out.push({
          dataset: 'CSV',
          llm: label === 1 ? 'ChatGPT' : 'Human',
          label,
          text: csvMatch[2].replace(/""/g, '"').trim(),
        });
      }
    }

    return out;
  }

  return [];
}

app.post('/api/mgtbench/detect', async (req, res) => {
  try {
    const {
      text = '',
      methodIds,
      cutTokenLength,
      threshold = 0.5,
      gptzeroApiKey = '',
    } = req.body || {};

    const cleanText =
      typeof text === 'string'
        ? text.trim()
        : '';

    if (cleanText.length < 10) {
      return res.status(400).json({
        error: 'Введите минимум 10 символов для анализа MGTBench.',
      });
    }

    const validMethodIds = Array.isArray(methodIds)
      ? methodIds.filter(
          (id: unknown): id is string =>
            typeof id === 'string'
        )
      : undefined;

    const cutLen =
      typeof cutTokenLength === 'number' &&
      Number.isFinite(cutTokenLength) &&
      cutTokenLength > 0
        ? Math.floor(cutTokenLength)
        : undefined;

    const thr =
      typeof threshold === 'number' &&
      Number.isFinite(threshold)
        ? Math.max(0, Math.min(1, threshold))
        : 0.5;

    const report: MgtBenchDetectReport =
      runMgtBenchDetection(cleanText, {
        methodIds: validMethodIds,
        cutTokenLength: cutLen,
        threshold: thr,
      });

    // Необязательное подключение к официальному API GPTZero.
    // Без API-ключа используется локальная эвристическая оценка.
    if (
      typeof gptzeroApiKey === 'string' &&
      gptzeroApiKey.trim()
    ) {
      try {
        const controller = new AbortController();

        const timer = setTimeout(
          () => controller.abort(),
          12_000
        );

        try {
          const gzResp = await fetch(
            'https://api.gptzero.me/v2/predict/text',
            {
              method: 'POST',
              headers: {
                'Content-Type': 'application/json',
                Accept: 'application/json',
                'X-Api-Key': gptzeroApiKey.trim(),
              },
              signal: controller.signal,
              body: JSON.stringify({
                document: report.textExcerpt,
              }),
            }
          );

          if (gzResp.ok) {
            const gzData: any = await gzResp.json();
            const doc = gzData?.documents?.[0];

            if (
              doc &&
              typeof doc.completely_generated_prob === 'number'
            ) {
              const liveProb = Number(
                Math.max(
                  0.002,
                  Math.min(0.998, doc.completely_generated_prob)
                ).toFixed(4)
              );

              const idx = report.results.findIndex(
                (result) => result.id === 'GPTZero'
              );

              if (idx >= 0) {
                report.results[idx] = {
                  ...report.results[idx],

                  rawCriterion: liveProb,
                  aiProbability: liveProb,

                  humanProbability: Number(
                    (1 - liveProb).toFixed(4)
                  ),

                  prediction:
                    liveProb >= thr
                      ? 'ai'
                      : 'human',

                  details:
                    `Live GPTZero API: P(AI)=${(liveProb * 100).toFixed(1)}%, ` +
                    `avg_ppl=${doc.average_generated_prob ?? '—'}`,
                };

                report.flaggedCount = report.results.filter(
                  (result) => result.prediction === 'ai'
                ).length;

                const avg =
                  report.results.reduce(
                    (acc, result) => acc + result.aiProbability,
                    0
                  ) /
                  Math.max(1, report.results.length);

                report.overallAiProbability = Number(
                  avg.toFixed(4)
                );

                report.overallHumanProbability = Number(
                  (1 - avg).toFixed(4)
                );

                report.overallVerdict =
                  avg >= 0.6
                    ? 'Machine-Generated (AI)'
                    : avg <= 0.4
                      ? 'Human-Written'
                      : 'Borderline / Mixed';
              }
            }
          }
        } finally {
          clearTimeout(timer);
        }
      } catch (gzErr: any) {
        console.warn(
          'Live GPTZero bridge fallback to MGTBench local engine:',
          gzErr?.message
        );
      }
    }

    mgtBenchStats.totalDetections += 1;
    res.json(report);
  } catch (err: any) {
    console.error('MGTBench detect error:', err);
    res.status(400).json({
      error: err?.message || 'Ошибка анализа MGTBench',
    });
  }
});

function parseDetectZooSamples(
  rawInput: unknown
): DetectZooSample[] {
  if (Array.isArray(rawInput)) {
    const out: DetectZooSample[] = [];

    for (let i = 0; i < rawInput.length; i++) {
      const item: any = rawInput[i];

      if (
        item &&
        typeof item.text === 'string' &&
        item.text.trim()
      ) {
        const label = Number(item.label) > 0.5 ? 1 : 0;

        out.push({
          id:
            typeof item.id === 'string'
              ? item.id
              : `custom_${i + 1}`,

          text: item.text.trim(),
          label,

          source:
            typeof item.source === 'string'
              ? item.source
              : label === 1
                ? 'AI'
                : 'Human',

          dataset:
            typeof item.dataset === 'string'
              ? item.dataset
              : 'custom',
        });
      }
    }

    return out;
  }

  if (
    typeof rawInput === 'string' &&
    rawInput.trim()
  ) {
    const lines = rawInput
      .split(/\r?\n/)
      .map((line) => line.trim())
      .filter(Boolean);

    const out: DetectZooSample[] = [];

    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];

      if (line.startsWith('{')) {
        try {
          const obj = JSON.parse(line);

          if (
            obj &&
            typeof obj.text === 'string' &&
            obj.text.trim()
          ) {
            const label = Number(obj.label) > 0.5 ? 1 : 0;

            out.push({
              id:
                typeof obj.id === 'string'
                  ? obj.id
                  : `jsonl_${i + 1}`,

              text: obj.text.trim(),
              label,

              source:
                typeof obj.source === 'string'
                  ? obj.source
                  : label === 1
                    ? 'AI'
                    : 'Human',

              dataset:
                typeof obj.dataset === 'string'
                  ? obj.dataset
                  : 'custom',
            });
          }

          continue;
        } catch {
          // При ошибке JSON пробуем разобрать строку как CSV/TSV.
        }
      }

      const separator = line.includes('\t')
        ? '\t'
        : ',';

      const firstSeparator = line.indexOf(separator);

      if (firstSeparator > 0) {
        const labelText = line
          .slice(0, firstSeparator)
          .trim()
          .toLowerCase();

        const text = line
          .slice(firstSeparator + 1)
          .trim();

        if (
          text &&
          text.toLowerCase() !== 'text'
        ) {
          const isAi =
            labelText === '1' ||
            labelText === 'ai' ||
            labelText === 'machine' ||
            labelText === 'llm' ||
            labelText === 'true';

          out.push({
            id: `row_${i + 1}`,
            text,
            label: isAi ? 1 : 0,
            source: isAi ? 'AI' : 'Human',
            dataset: 'custom',
          });
        }
      }
    }

    return out;
  }

  return [];
}

        error: 'Введите минимум 10 символов для анализа DetectZoo.',
      });
    }

    const validNames = Array.isArray(detectorNames)
      ? detectorNames.filter(
          (name: unknown): name is string =>
            typeof name === 'string'
        )
      : undefined;

    const threshold =
      typeof thresholdOverride === 'number' &&
      Number.isFinite(thresholdOverride)
        ? Math.max(0, Math.min(1, thresholdOverride))
        : undefined;

    const normalizedMode =
      mode === 'fast'
        ? 'fast'
        : validNames && validNames.length > 0
          ? 'selected'
          : 'full';

    const report: DetectZooSuiteReport =
      runDetectZooTextSuite(cleanText, {
        detectorNames: validNames,
        thresholdOverride: threshold,
        mode: normalizedMode,
      });

    detectZooStats.totalPredictions += 1;

    res.json(report);
  } catch (err: any) {
    console.error('DetectZoo predict error:', err);
    res.status(400).json({
      error: err?.message || 'Ошибка анализа DetectZoo',
    });
  }
});

function analyzeTextMetrics(text: string) {
  const rawSentences = (
    text.match(/[^.!?…]+(?:[.!?…]+|$)/g) ?? []
  )
    .map((sentence: string) => sentence.trim())
    .filter((sentence: string) => sentence.length > 2);

  const sentenceWordCounts = rawSentences.map(
    (sentence) =>
      sentence.split(/\s+/).filter(Boolean).length
  );

  const totalWords =
    sentenceWordCounts.reduce(
      (sum: number, count: number) => sum + count,
      0
    ) || 1;

  const avgWords =
    totalWords / (rawSentences.length || 1);

  const variance =
    sentenceWordCounts.reduce(
      (sum: number, length: number) =>
        sum + Math.pow(length - avgWords, 2),
      0
    ) /
    (rawSentences.length || 1);

  const stdDev = Math.sqrt(variance);

  const burstiness =
    avgWords > 0
      ? Number((stdDev / avgWords).toFixed(2))
      : 0;

  const words =
    text.toLowerCase().match(/[а-яёa-z0-9-]+/gi) || [];

  const uniqueWords = new Set(words);

  const ttr =
    words.length > 0
      ? Number((uniqueWords.size / words.length).toFixed(2))
      : 0.5;

  const lower = text.toLowerCase();

  const foundCliches = EXTENDED_AI_CLICHES.filter(
    (cliche: string) => lower.includes(cliche)
  );

  const clichePenalty = foundCliches.length * 7;

  const burstinessBonus = Math.min(
    18,
    Math.max(
      -25,
      Math.round((burstiness - 0.45) * 40)
    )
  );

  const ttrBonus = Math.min(
    10,
    Math.max(
      -15,
      Math.round((ttr - 0.55) * 35)
    )
  );

  const baseHuman = Math.max(
    12,
    Math.min(
      99,
      96 - clichePenalty + burstinessBonus + ttrBonus
    )
  );

  const top10 = Math.max(
    22,
    Math.min(
      68,
      Math.round(
        52 -
        burstiness * 18 -
        (foundCliches.length === 0
          ? 8
          : -foundCliches.length * 4)
      )
    )
  );

  const top100 = Math.max(
    20,
    Math.min(
      38,
      Math.round(31 - foundCliches.length)
    )
  );

  const top1000 = Math.max(
    10,
    Math.min(
      26,
      Math.round(14 + burstiness * 6)
    )
  );

  const outOf1000 = Math.max(
    4,
    100 - (top10 + top100 + top1000)
  );

  const binocularsScore = Math.max(
    0.795,
    Math.min(
      0.985,
      Number(
        (
          0.835 +
          foundCliches.length * 0.035 -
          burstiness * 0.05
        ).toFixed(3)
      )
    )
  );

  const detectGptCurvature = Math.max(
    0.012,
    Math.min(
      0.42,
      Number(
        (
          0.045 +
          foundCliches.length * 0.04 -
          (burstiness > 0.6 ? 0.025 : 0)
        ).toFixed(3)
      )
    )
  );

  /*
   * ВАЖНО:
   * Следующие показатели получаются из локальных ядер проекта.
   * Они не означают, что здесь выполняется полноценный инференс
   * оригинальных моделей Binoculars, DetectGPT, GLTR и других
   * сторонних детекторов.
   *
   * Это эвристические оценки, предназначенные для внутренней
   * диагностики и сравнительного анализа.
   */

  const slopReport = runSlopTotalAnalysis(text);

  const slopHuman = Math.max(
    5,
    Math.min(
      99,
      Math.round(slopReport.humanScore)
    )
  );

  const slopAi = Math.max(
    1,
    Math.min(
      95,
      Math.round(slopReport.overallScore)
    )
  );

  const mgtReport = runMgtEvalDetection(text);

  const mgtHuman = Math.max(
    5,
    Math.min(
      99,
      Math.round(
        mgtReport.overallHumanProbability * 100
      )
    )
  );

  const mgtAi = Math.max(
    1,
    Math.min(
      95,
      Math.round(
        mgtReport.overallAiProbability * 100
      )
    )
  );

  const mgtBenchReport = runMgtBenchDetection(text);

  const mgtBenchHuman = Math.max(
    5,
    Math.min(
      99,
      Math.round(
        mgtBenchReport.overallHumanProbability * 100
      )
    )
  );

  const mgtBenchAi = Math.max(
    1,
    Math.min(
      95,
      Math.round(
        mgtBenchReport.overallAiProbability * 100
      )
    )
  );

  const detectZooReport = runDetectZooTextSuite(text);

  const detectZooHuman = Math.max(
    5,
    Math.min(
      99,
      Math.round(
        detectZooReport.overallHumanProbability * 100
      )
    )
  );

  const detectZooAi = Math.max(
    1,
    Math.min(
      95,
      Math.round(
        detectZooReport.overallAiProbability * 100
      )
    )
  );

  const detectorsMap: Record<
    string,
    {
      rawMetric: string;
      threshold: string;
      scoreDelta: number;
      overrideHuman?: number;
      overrideAi?: number;
    }
  > = {
    binoculars: {
      rawMetric: `${binocularsScore}`,
      threshold: '< 0.901',
      scoreDelta:
        binocularsScore < 0.901
          ? 2
          : -20,
    },

    detectGpt: {
      rawMetric: `Z = ${detectGptCurvature}`,
      threshold: 'Z < 0.15',
      scoreDelta:
        detectGptCurvature < 0.15
          ? 1
          : -18,
    },

    fastDetectGpt: {
      rawMetric:
        `P = ${Number(
          Math.min(0.99, baseHuman / 100 + 0.02).toFixed(2)
        )}`,
      threshold: 'P > 0.85',
      scoreDelta: 2,
    },

    gltr: {
      rawMetric: `Top-10: ${top10}%`,
      threshold: '< 45%',
      scoreDelta:
        top10 < 45
          ? 1
          : -15,
    },

    radar: {
      rawMetric:
        `Conf: ${Number(
          Math.min(0.99, baseHuman / 100).toFixed(2)
        )}`,
      threshold: 'Conf > 0.80',
      scoreDelta: 1,
    },

    ghostbuster: {
      rawMetric:
        `Diverg: ${Number(
          Math.max(
            0.02,
            0.14 - (baseHuman / 100) * 0.1
          ).toFixed(3)
        )}`,
      threshold: '< 0.12',
      scoreDelta: 1,
    },

    sloptotal: {
      rawMetric:
        `${slopReport.overallScore}/100 ` +
        `(${slopReport.enginesFlagged}/${slopReport.enginesTotal})`,
      threshold: '< 30 (Clean)',
      scoreDelta: 0,
      overrideHuman: slopHuman,
      overrideAi: slopAi,
    },

    mgtEval: {
      rawMetric:
        `${mgtAi}% AI ` +
        `(${mgtReport.flaggedCount}/${mgtReport.totalRun})`,
      threshold: '< 50% P(AI)',
      scoreDelta: 0,
      overrideHuman: mgtHuman,
      overrideAi: mgtAi,
    },

    mgtBench: {
      rawMetric:
        `${mgtBenchAi}% AI ` +
        `(${mgtBenchReport.flaggedCount}/${mgtBenchReport.totalRun})`,
      threshold: '< 50% P(AI)',
      scoreDelta: 0,
      overrideHuman: mgtBenchHuman,
      overrideAi: mgtBenchAi,
    },

    detectZoo: {
      rawMetric:
        `${detectZooAi}% AI ` +
        `(${detectZooReport.flaggedCount}/${detectZooReport.totalRun})`,
      threshold: '< 50% P(AI)',
      scoreDelta: 0,
      overrideHuman: detectZooHuman,
      overrideAi: detectZooAi,
    },

    robertaOpenAi: {
      rawMetric: `Fake: ${Math.max(1, 100 - baseHuman)}%`,
      threshold: '< 10%',
      scoreDelta: 0,
    },

    desklib: {
      rawMetric:
        `Anomaly: ${Math.max(
          1,
          Math.round((100 - baseHuman) * 0.7)
        )}%`,
      threshold: '< 8%',
      scoreDelta: 2,
    },

    aiTextJury: {
      rawMetric:
        baseHuman >= 80
          ? '5/5 Человек'
          : baseHuman >= 60
            ? '3/5 Человек'
            : '1/5 Человек',
      threshold: '? 4/5',
      scoreDelta:
        baseHuman >= 80
          ? 1
          : -10,
    },

    hc3Comparison: {
      rawMetric:
        `GPT-Sim: ${Math.max(
          1,
          Math.round((100 - baseHuman) * 0.8)
        )}%`,
      threshold: '< 5%',
      scoreDelta: 1,
    },

    houtini: {
      rawMetric:
        `Sym: ${Number(
          Math.max(
            0.04,
            0.25 - burstiness * 0.15
          ).toFixed(2)
        )}`,
      threshold: '< 0.20',
      scoreDelta: 2,
    },

    detectLlama: {
      rawMetric: 'н/д (эвристика)',
      threshold: '< 5%',
      scoreDelta: 3,
    },
  };

  const results = Object.keys(detectorsMap).map((id) => {
    const meta = detectorsMap[id];

    const humanScore =
      typeof meta.overrideHuman === 'number'
        ? meta.overrideHuman
        : Math.max(
            8,
            Math.min(
              99,
              Math.round(baseHuman + meta.scoreDelta)
            )
          );

    const aiScore =
      typeof meta.overrideAi === 'number'
        ? meta.overrideAi
        : 100 - humanScore;

    return {
      id,
      humanScore,
      aiScore,

      rawMetric: meta.rawMetric,
      threshold: meta.threshold,

      status:
        humanScore >= 80
          ? 'Пройден (Человек)'
          : 'Подозрение на ИИ',

      verdictLabel:
        humanScore >= 80
          ? 'Человек (Аутентичный)'
          : humanScore >= 60
            ? 'Смешанный (Граница)'
            : 'ИИ (Сгенерирован)',

      riskLevel:
        humanScore >= 80
          ? 'low'
          : humanScore >= 60
            ? 'medium'
            : 'high',
    };
  });

  const avgHuman = Math.round(
    results.reduce(
      (acc, result) => acc + result.humanScore,
      0
    ) / results.length
  );

  const flaggedCount = results.filter(
    (result) => result.humanScore < 70
  ).length;

  return {
    heuristic: true,

    auditedWords: words.length,

    overallHumanScore: avgHuman,
    overallAiScore: 100 - avgHuman,

    flaggedCount,
    totalAudited: results.length,

    verdict:
      avgHuman >= 85
        ? 'Эвристика: признаков машинного текста мало'
        : avgHuman >= 65
          ? 'Эвристика: умеренный риск'
          : 'Эвристика: много признаков машинного текста',

    gltrDistribution: {
      top10,
      top100,
      top1000,
      outOf1000,
    },

    binocularsScore: binocularsScore.toString(),
    detectGptCurvature: detectGptCurvature.toString(),

    sloptotalSummary: {
      overallScore: slopReport.overallScore,
      overallVerdict: slopReport.overallVerdict,
      confidence: slopReport.confidence,
      enginesFlagged: slopReport.enginesFlagged,
      enginesTotal: slopReport.enginesTotal,
      language: slopReport.language,
    },

    mgtEvalSummary: {
      overallAiPct: mgtAi,
      flaggedCount: mgtReport.flaggedCount,
      totalRun: mgtReport.totalRun,
      overallVerdict: mgtReport.overallVerdict,
    },

    mgtBenchSummary: {
      overallAiPct: mgtBenchAi,
      flaggedCount: mgtBenchReport.flaggedCount,
      totalRun: mgtBenchReport.totalRun,
      overallVerdict: mgtBenchReport.overallVerdict,
      phdIntrinsicDim: mgtBenchReport.phdIntrinsicDim,
    },

    detectZooSummary: {
      overallAiPct: detectZooAi,
      flaggedCount: detectZooReport.flaggedCount,
      totalRun: detectZooReport.totalRun,
      overallVerdict: detectZooReport.overallVerdict,
      meanConfidence: detectZooReport.meanConfidence,
      phdIntrinsicDim:
        detectZooReport.specialSignals.phdIntrinsicDim,
    },

    results,
  };
}

app.post('/api/audit-detectors', async (req, res) => {
  try {
    const {
      text = '',
      detectorId,
      detectorIds,
    } = req.body || {};

    if (
      typeof text !== 'string' ||
      !text.trim()
    ) {
      return res.status(400).json({
        error: 'Нет текста для аудита',
      });
    }

    const auditData = analyzeTextMetrics(text);

    if (detectorId) {
      const single = auditData.results.find(
        (result) => result.id === detectorId
      );

      if (single) {
        return res.json({
          heuristic: true,
          singleDetector: single,
          auditedWords: auditData.auditedWords,
          overallHumanScore: auditData.overallHumanScore,
        });
      }
    }

    if (
      Array.isArray(detectorIds) &&
      detectorIds.length > 0
    ) {
      const allowed = new Set(
        detectorIds.filter(
          (id: unknown): id is string =>
            typeof id === 'string'
        )
      );

      const filteredResults = auditData.results.filter(
        (result) => allowed.has(result.id)
      );

      const subsetAvgHuman =
        filteredResults.length > 0
          ? Math.round(
              filteredResults.reduce(
                (acc, result) => acc + result.humanScore,
                0
              ) / filteredResults.length
            )
          : auditData.overallHumanScore;

      const subsetFlagged = filteredResults.filter(
        (result) => result.humanScore < 70
      ).length;

      return res.json({
        ...auditData,

        selectedMode: true,

        totalAudited: filteredResults.length,

        overallHumanScore: subsetAvgHuman,
        overallAiScore: 100 - subsetAvgHuman,

        flaggedCount: subsetFlagged,

        verdict:
          subsetAvgHuman >= 85
            ? 'Эвристика (выбранные): признаков машинного текста мало'
            : subsetAvgHuman >= 65
              ? 'Эвристика (выбранные): умеренный риск'
              : 'Эвристика (выбранные): много признаков машинного текста',

        results: filteredResults,
      });
    }

    res.json(auditData);
  } catch (err: any) {
    console.error('Audit error:', err);

    res.status(500).json({
      error: err?.message || 'Ошибка аудита',
    });
  }
});

export function stemRussianWord(raw: string): string {
  const w = raw.toLowerCase().replace(/ё/g, 'е').replace(/[^а-яa-z0-9-]/g, '');
  if (w.length <= 4) return w;

  if (/^[a-z0-9-]+$/.test(w)) {
    return w.replace(/(?:ing|tion|ment|ness|able|ible|ed|es|s)$/, '') || w;
  }

  let s = w;
  s = s.replace(/(?:ться|ться|вши|вшись|ивши|ившись|ывши|ывшись)$/, '');
  s = s.replace(
    /(?:иями|ями|ами|ого|ему|ому|ыми|ими|его|ее|ие|ые|ой|ий|ый|ая|яя|ое|ую|юю|им|ым|ем|ом|ам|ям|ах|ях|ов|ев|ей|ию|ью|ия|ья|ие|ье|у|ю|а|я|е|о|и|ы)$/,
    ''
  );

  if (s.length >= 6) {
    s = s.replace(/(?:ующ|ющ|ащ|ящ|вш|енн|анн|онн|ост|еств|ирова|ова|ева)$/, '');
  }

  return s.length >= 3 ? s : w.slice(0, Math.min(4, w.length));
}

function splitParagraph(paragraph: string): string[] {
  const raw = paragraph
    .split(/(?<=[.!?…])\s+(?=[А-ЯЁA-Z«"(\d—–-])/u)
    .map((s) => s.trim())
    .filter(Boolean);

  const merged: string[] = [];

  for (const s of raw) {
    if (merged.length > 0 && ABBREVIATION_END.test(merged[merged.length - 1])) {
      merged[merged.length - 1] += ' ' + s;
    } else {
      merged.push(s);
    }
  }

  return merged.filter((s) => s.length > 2);
}

export function splitSentences(text: string): string[] {
  return text.split(/\n{2,}/).flatMap(splitParagraph);
}

export function getSentenceWords(sentence: string): string[] {
  return sentence.match(/[а-яёa-z0-9-]+/gi) || [];
}

export function movingTTR(words: string[], windowSize = 50): number {
  const stemCache = new Map<string, string>();
  const stemmed: string[] = [];

  for (const word of words) {
    let stem = stemCache.get(word);

    if (stem === undefined) {
      stem = stemRussianWord(word);
      stemCache.set(word, stem);
    }

    if (stem.length >= 2) stemmed.push(stem);
  }

  const n = stemmed.length;
  if (n === 0) return 0.65;

  const safeWindow = Math.max(1, Math.floor(windowSize));

  if (n < safeWindow) {
    return Number((new Set(stemmed).size / n).toFixed(2));
  }

  const frequencies = new Map<string, number>();
  let unique = 0;

  for (let i = 0; i < safeWindow; i++) {
    const word = stemmed[i];
    const next = (frequencies.get(word) || 0) + 1;
    frequencies.set(word, next);
    if (next === 1) unique++;
  }

  const step = Math.max(1, Math.floor(safeWindow / 5));
  let sum = unique / safeWindow;
  let count = 1;
  let windowStart = 0;

  while (windowStart + step + safeWindow <= n) {
    for (let j = windowStart; j < windowStart + step; j++) {
      const word = stemmed[j];
      const next = (frequencies.get(word) || 0) - 1;

      if (next <= 0) {
        frequencies.delete(word);
        unique--;
      } else {
        frequencies.set(word, next);
      }
    }

    const nextStart = windowStart + step;

    for (let j = windowStart + safeWindow; j < nextStart + safeWindow; j++) {
      const word = stemmed[j];
      const next = (frequencies.get(word) || 0) + 1;
      frequencies.set(word, next);
      if (next === 1) unique++;
    }

    windowStart = nextStart;
    sum += unique / safeWindow;
    count++;
  }

  return Number((sum / count).toFixed(2));
}

function getOpenerPattern(sentence: string): string {
  const trimmed = sentence.trim().replace(/^[«"(\-—–]+/, '');
  const words = getSentenceWords(trimmed);

  if (words.length === 0) return '';

  const w0 = words[0].toLowerCase();
  const w1 = (words[1] || '').toLowerCase();

  if (/^(в|во|на|при|для|через|после|среди|между|из|от|к|по|о|об)$/.test(w0)) {
    if (/(?:ой|ей|ом|ем|ых|их|ого|его)$/.test(w1)) return `PREP+ADJ(${w0})`;
    return `PREP(${w0})`;
  }

  if (/^(однако|кроме|более|таким|следовательно|тем|причем|причём|впрочем)$/.test(w0)) {
    return `INTRO(${w0})`;
  }

  return `W(${stemRussianWord(w0)})`;
}

// Сигнал считается авторской особенностью не по одному совпадению,
// а по повторяемости, распределению по абзацам и числу примеров.
// Низкая уверенность означает: конструкцию лучше не исправлять автоматически.
export function extractAuthorialDeviationProfile(
  donorText: string
): AuthorialDeviationProfile {
  const paragraphs = donorText
    .split(/\n{2,}/)
    .map((p) => p.trim())
    .filter(Boolean);

  const sentences = paragraphs.flatMap(splitParagraph);
  const totalWords = getSentenceWords(donorText).length;

  const definitions: Array<{
    kind: AuthorialDeviationKind;
    regex: RegExp;
  }> = [
    { kind: 'genitiveChain', regex: GENITIVE_CHAIN_REGEX },
    { kind: 'copula', regex: COPULA_REGEX },
    { kind: 'symmetry', regex: /не только[\s\S]{1,180}?но и/i },
    { kind: 'triad', regex: TRIAD_REGEX },
  ];

  const signals: AuthorialDeviationSignal[] = [];

  for (const definition of definitions) {
    const examples: string[] = [];
    const paragraphIndexes = new Set<number>();
    let count = 0;

    paragraphs.forEach((paragraph, paragraphIndex) => {
      const paragraphSentences = splitParagraph(paragraph);

      for (const sentence of paragraphSentences) {
        definition.regex.lastIndex = 0;

        if (definition.regex.test(sentence)) {
          count++;
          paragraphIndexes.add(paragraphIndex);

          if (examples.length < 4) {
            examples.push(sentence.trim());
          }
        }
      }
    });

    if (count === 0) continue;

    const per300Words = Number(((count / Math.max(1, totalWords)) * 300).toFixed(2));
    const paragraphSpread =
      paragraphs.length > 0
        ? Number((paragraphIndexes.size / paragraphs.length).toFixed(2))
        : 0;

    const repetitionFactor = Math.min(1, count / 3);
    const spreadFactor = Math.min(1, paragraphSpread / 0.5);
    const confidence = Number(
      Math.min(0.98, 0.2 + repetitionFactor * 0.45 + spreadFactor * 0.33).toFixed(2)
    );

    signals.push({
      kind: definition.kind,
      count,
      per300Words,
      paragraphSpread,
      confidence,
      examples,
    });
  }

  // Короткие самостоятельные фразы: фиксируем только повторяющийся приём,
  // а не единичную точку или случайный обрыв.
  const shortFragments = sentences.filter((sentence) => {
    const words = getSentenceWords(sentence);
    return words.length >= 1 && words.length <= 5;
  });

  if (shortFragments.length >= 2) {
    const examples = shortFragments.slice(0, 4);
    const per300Words = Number(
      ((shortFragments.length / Math.max(1, totalWords)) * 300).toFixed(2)
    );

    signals.push({
      kind: 'shortFragment',
      count: shortFragments.length,
      per300Words,
      paragraphSpread: 0,
      confidence: Number(Math.min(0.9, 0.25 + shortFragments.length * 0.12).toFixed(2)),
      examples,
    });
  }

  // Повторяющиеся необычные зачины: учитываем только одинаковые каркасы,
  // встречающиеся неоднократно, а не обычные союзы и предлоги.
  const openerCounts = new Map<string, { count: number; examples: string[] }>();

  for (const sentence of sentences) {
    const pattern = getOpenerPattern(sentence);

    if (!pattern || pattern.startsWith('PREP(') || pattern.startsWith('PREP+ADJ(')) {
      continue;
    }

    const current = openerCounts.get(pattern) || { count: 0, examples: [] };
    current.count++;

    if (current.examples.length < 3) {
      current.examples.push(sentence);
    }

    openerCounts.set(pattern, current);
  }

  const repeatedOpeners = Array.from(openerCounts.values())
    .filter((item) => item.count >= 3)
    .sort((a, b) => b.count - a.count);

  if (repeatedOpeners.length > 0) {
    const count = repeatedOpeners.reduce((sum, item) => sum + item.count, 0);
    const examples = repeatedOpeners.flatMap((item) => item.examples).slice(0, 4);

    signals.push({
      kind: 'unusualOpening',
      count,
      per300Words: Number(((count / Math.max(1, totalWords)) * 300).toFixed(2)),
      paragraphSpread: 0,
      confidence: Number(Math.min(0.9, 0.3 + count * 0.08).toFixed(2)),
      examples,
    });
  }

  const highConfidenceSignals = signals.filter((signal) => signal.confidence >= 0.72).length;
  const protectionLevel: AuthorialDeviationProfile['protectionLevel'] =
    highConfidenceSignals >= 3
      ? 'high'
      : highConfidenceSignals >= 1 || signals.length >= 3
        ? 'medium'
        : 'low';

  return {
    totalWords,
    analyzedParagraphs: paragraphs.length,
    signals,
    protectionLevel,
  };
}

const NEGATION_CUES_REGEX = /(^|[^а-яё])(не|нет|ни|нельзя|никогда|никуда|ничуть|никак|отнюдь|вопреки|несмотря на|лишь|только)(?![а-яё])/gi;
const MODAL_CUES_LIST = [
  'возможно', 'вероятно', 'может', 'должен', 'должна', 'должно', 'должны',
  'необходимо', 'следует', 'всегда', 'редко', 'иногда', 'запрещено', 'нельзя', 'допустимо',
];

const ABSTRACT_NOMINAL_REGEX = /^[а-яё]{5,}(?:ение|ание|ость|еств|ство|ция|ции|изм|изма|ика|ения|ания|ости|ством|цией|ений|аний)$/i;
const PARTICIPLE_CLAUSE_REGEX = /,\s*[а-яё]{4,}(?:ющ|ящ|ущ|ащ|вш|ш|енн|анн|нн|ем|им)[а-яё]{1,4}(?:\s|,|$)|,\s*[а-яё]{4,}(?:я|а|в|вши|вшись)(?=\s+[а-яё])/gi;
const MODAL_PARTICLE_REGEX = /(^|[^а-яё])(ведь|же|ли|бы|уж|разве|неужели|будто|словно|вряд ли|едва ли|разве что|мол|дескать|пожалуй|чай)(?![а-яё])/gi;
const PERSON_PRONOUN_REGEX = /(^|[^а-яё])(я|мы|нам|нас|наш|наша|наше|наши|мне|мной|вы|вам|вас|ты|тебе)(?![а-яё])/gi;

function computeMorphosyntaxStats(text: string, sentences: string[], words: string[]) {
  const safeWords = Math.max(1, words.length);
  const safeSents = Math.max(1, sentences.length);
  const charLenK = Math.max(0.5, text.length / 1000);

  let nominals = 0;
  for (const w of words) {
    if (ABSTRACT_NOMINAL_REGEX.test(w)) nominals++;
  }
  const nominalizationPer100 = Number(((nominals / safeWords) * 100).toFixed(1));

  let participleHits = 0;
  for (const s of sentences) {
    const m = s.match(PARTICIPLE_CLAUSE_REGEX);
    if (m) participleHits += m.length;
  }
  const participleClausePerSent = Number((participleHits / safeSents).toFixed(2));

  const modalMatches = text.match(MODAL_PARTICLE_REGEX) || [];
  const modalParticlePer1k = Number((modalMatches.length / charLenK).toFixed(1));

  const personMatches = text.match(PERSON_PRONOUN_REGEX) || [];
  const firstSecondPersonPer1k = Number((personMatches.length / charLenK).toFixed(1));

  return {
    nominalizationPer100,
    participleClausePerSent,
    modalParticlePer1k,
    firstSecondPersonPer1k,
  };
}

export function extractFactInventory(sourceText: string): FactInventory {
  const numbersAndDates = Array.from(
    new Set(
      (sourceText.match(/\b\d[\d.,–-]*\d\b|\b\d+\b|\b[IVXLCDM]{2,6}\b/g) ?? []).filter(
        (n) => n.length >= 1 && !/^(II|III)$/.test(n)
      )
    )
  );

  const sentences = splitSentences(sourceText);
  const properSet = new Set<string>();

  for (const s of sentences) {
    const tokens = s.match(/[А-ЯЁA-Z][а-яёa-zА-ЯЁA-Z-]{2,}/g) ?? [];

    tokens.forEach((tok, idx) => {
      const lower = tok.toLowerCase();

      if (idx === 0 && (SENTENCE_START_STOPWORDS.has(lower) || RUSSIAN_FUNCTION_WORDS.has(lower))) {
        return;
      }

      if (idx > 0 || tok.length >= 5) {
        if (!SENTENCE_START_STOPWORDS.has(lower)) {
          properSet.add(tok);
        }
      }
    });
  }

  const wordFreq = new Map<string, { raw: string; count: number }>();
  const allWords = (sourceText.toLowerCase().match(/[а-яё]{7,}/g) ?? []).filter(
    (w) => !RUSSIAN_FUNCTION_WORDS.has(w) && !SENTENCE_START_STOPWORDS.has(w)
  );

  for (const w of allWords) {
    const st = stemRussianWord(w);
    if (st.length < 5) continue;

    const prev = wordFreq.get(st);
    wordFreq.set(st, { raw: prev ? prev.raw : w, count: (prev?.count || 0) + 1 });
  }

  const domainTerms = Array.from(wordFreq.values())
    .sort((a, b) => b.count - a.count)
    .slice(0, 14)
    .map((x) => x.raw);

  const negationCount = (sourceText.match(NEGATION_CUES_REGEX) || []).length;
  const lowerSrc = sourceText.toLowerCase();
  const modalCues = MODAL_CUES_LIST.filter((m) =>
    new RegExp(`(^|[^а-яё])${m}(?![а-яё])`, 'i').test(lowerSrc)
  );

  return {
    numbersAndDates,
    properNames: Array.from(properSet),
    domainTerms,
    negationCount,
    modalCues,
  };
}

export function extractDonorForbiddenTerms(donorText: string, contentText: string): string[] {
  if (!donorText.trim() || !contentText.trim()) return [];
  const donorInv = extractFactInventory(donorText);
  const contentLower = contentText.toLowerCase().replace(/ё/g, 'е');
  const contentStems = new Set(
    (contentLower.match(/[а-яёa-z0-9-]+/gi) ?? []).map(stemRussianWord)
  );

  const forbidden: string[] = [];
  const seenNorm = new Set<string>();
  const candidates = [...donorInv.properNames, ...donorInv.domainTerms];
  for (const term of candidates) {
    const norm = term.toLowerCase().replace(/ё/g, 'е');
    const st = stemRussianWord(term);
    if (
      norm.length >= 4 &&
      !seenNorm.has(norm) &&
      !contentLower.includes(norm) &&
      !contentStems.has(st)
    ) {
      seenNorm.add(norm);
      forbidden.push(term);
    }
  }
  return forbidden.slice(0, 24);
}

export function verifyFactsPreserved(
  sourceText: string,
  outputText: string,
  donorProfile?: DonorDNAProfile
): FactVerificationReport {
  if (!sourceText.trim()) {
    return { preservedPct: 100, lostCritical: [], lostTerms: [], negationDelta: 0, lostModals: [], donorLeakTerms: [] };
  }

  const inv = extractFactInventory(sourceText);
  const outLower = outputText.toLowerCase().replace(/ё/g, 'е');
  const outputTokens = outLower.match(/[а-яёa-z0-9-]+/gi) ?? [];
  const outStems = new Set(outputTokens.map(stemRussianWord));

  const lostCritical: string[] = [];

  for (const num of inv.numbersAndDates) {
    if (!outputText.includes(num)) {
      lostCritical.push(num);
    }
  }

  for (const name of inv.properNames) {
    const st = stemRussianWord(name);
    const normalizedName = name.toLowerCase().replace(/ё/g, 'е');
    const matched = outLower.includes(normalizedName) || outStems.has(st);

    if (!matched) {
      lostCritical.push(name);
    }
  }

  const lostTerms: string[] = [];

  for (const term of inv.domainTerms) {
    const st = stemRussianWord(term);
    const matched = outStems.has(st);

    if (!matched) {
      lostTerms.push(term);
    }
  }

  const outNegations = (outputText.match(NEGATION_CUES_REGEX) || []).length;
  const negationDelta = inv.negationCount - outNegations;
  const lostModals = inv.modalCues.filter(
    (m) => !new RegExp(`(^|[^а-яё])${m}(?![а-яё])`, 'i').test(outLower)
  );

  // Check if donor-specific topic terms leaked into output (case-insensitive dedup)
  const donorLeakTerms: string[] = [];
  const seenLeakNorm = new Set<string>();
  if (donorProfile && donorProfile.donorTopicTerms.length > 0) {
    const sourceLower = sourceText.toLowerCase().replace(/ё/g, 'е');
    const sourceStems = new Set((sourceLower.match(/[а-яёa-z0-9-]+/gi) ?? []).map(stemRussianWord));
    for (const dTerm of donorProfile.donorTopicTerms) {
      const dNorm = dTerm.toLowerCase().replace(/ё/g, 'е');
      if (seenLeakNorm.has(dNorm)) continue;
      const dStem = stemRussianWord(dTerm);
      if (sourceLower.includes(dNorm) || sourceStems.has(dStem)) continue;
      if (outLower.includes(dNorm) || (dStem.length >= 5 && outStems.has(dStem))) {
        seenLeakNorm.add(dNorm);
        donorLeakTerms.push(dTerm);
      }
    }
  }

  const totalChecked = Math.max(
    1,
    inv.numbersAndDates.length + inv.properNames.length + inv.domainTerms.length
  );

  const lostWeighted = lostCritical.length + lostTerms.length * 0.5;
  const preservedPct = Math.max(
    0,
    Math.min(100, Math.round(((totalChecked - lostWeighted) / totalChecked) * 100))
  );

  return {
    preservedPct,
    lostCritical: Array.from(new Set(lostCritical)),
    lostTerms: Array.from(new Set(lostTerms)),
    negationDelta,
    lostModals,
    donorLeakTerms,
  };
}

function computeLengthHistogram(lengths: number[]): [number, number, number, number, number] {
  const bins: [number, number, number, number, number] = [0, 0, 0, 0, 0];
  const n = Math.max(1, lengths.length);
  for (const l of lengths) {
    if (l <= 5) bins[0]++;
    else if (l <= 10) bins[1]++;
    else if (l <= 16) bins[2]++;
    else if (l <= 24) bins[3]++;
    else bins[4]++;
  }
  return [
    Math.round((bins[0] / n) * 100),
    Math.round((bins[1] / n) * 100),
    Math.round((bins[2] / n) * 100),
    Math.round((bins[3] / n) * 100),
    Math.round((bins[4] / n) * 100),
  ];
}

function mineOpenersAndConnectors(sentences: string[]): {
  autoMinedOpeners: string[];
  autoMinedConnectors: string[];
} {
  const openerFreq = new Map<string, number>();
  const connFreq = new Map<string, number>();

  for (const s of sentences) {
    const words = getSentenceWords(s).map((w) => w.toLowerCase());
    if (words.length >= 2) {
      const bi = `${words[0]} ${words[1]}`;
      if (words[0].length >= 2 && !/^\d+$/.test(words[0])) {
        openerFreq.set(bi, (openerFreq.get(bi) || 0) + 1);
      }
    }
    // Extract intra-sentence connectors following punctuation (, / — / ;)
    const midMatches = s.match(/(?:,|—|;)\s+([а-яё]{2,12}(?:\s+[а-яё]{2,10})?)/gi) || [];
    for (const rawM of midMatches) {
      const clean = rawM.replace(/^[,—;\s]+/, '').toLowerCase();
      if (clean.length >= 3 && !RUSSIAN_FUNCTION_WORDS.has(clean)) {
        connFreq.set(clean, (connFreq.get(clean) || 0) + 1);
      }
    }
  }

  const autoMinedOpeners = Array.from(openerFreq.entries())
    .sort((a, b) => b[1] - a[1])
    .slice(0, 6)
    .map(([k]) => k);

  const autoMinedConnectors = Array.from(connFreq.entries())
    .filter(([, count]) => count >= 2 || sentences.length <= 8)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 8)
    .map(([k]) => k);

  return { autoMinedOpeners, autoMinedConnectors };
}

export function extractDonorDNAProfile(donorText: string): DonorDNAProfile {
  const paragraphs = donorText
    .split(/\n{2,}/)
    .map((p) => p.trim())
    .filter(Boolean);

  const paragraphSentences = paragraphs.map(splitParagraph);
  const paraSentenceCounts = paragraphSentences.map((items) => items.length).filter((n) => n > 0);
  const safeParaCounts = paraSentenceCounts.length > 0 ? paraSentenceCounts : [4];

  const avgSentPerPara = Number(
    (safeParaCounts.reduce((a, b) => a + b, 0) / safeParaCounts.length).toFixed(1)
  );

  const minSentPerPara = Math.min(...safeParaCounts);
  const maxSentPerPara = Math.max(...safeParaCounts);

  const sentences = paragraphSentences.flat();
  const sentenceWordLists = sentences.map(getSentenceWords);
  const lengths = sentenceWordLists.map((items) => items.length).filter((n) => n > 0);
  const safeLengths = lengths.length > 0 ? lengths : [12];
  const sorted = [...safeLengths].sort((a, b) => a - b);

  const totalWords = safeLengths.reduce((a, b) => a + b, 0) || 1;
  const avgLen = Number((totalWords / safeLengths.length).toFixed(1));
  const minLen = sorted[0];
  const maxLen = sorted[sorted.length - 1];
  const medianLen = sorted[Math.floor(sorted.length / 2)];

  const variance =
    safeLengths.reduce((sum, len) => sum + Math.pow(len - avgLen, 2), 0) / safeLengths.length;

  const stdDev = Number(Math.sqrt(variance).toFixed(1));
  const burstiness = avgLen > 0 ? Number((stdDev / avgLen).toFixed(2)) : 0.5;

  const shortCount = safeLengths.filter((n) => n <= 8).length;
  const longCount = safeLengths.filter((n) => n >= 22).length;

  const shortRatioPct = Math.round((shortCount / safeLengths.length) * 100);
  const longRatioPct = Math.round((longCount / safeLengths.length) * 100);

  const lengthHistogramPct = computeLengthHistogram(safeLengths);
  let deltaSum = 0;
  for (let i = 0; i < safeLengths.length - 1; i++) {
    deltaSum += Math.abs(safeLengths[i + 1] - safeLengths[i]);
  }
  const cadenceDeltaMean =
    safeLengths.length > 1 ? Number((deltaSum / (safeLengths.length - 1)).toFixed(1)) : 6.0;

  const allWords = donorText.toLowerCase().match(/[а-яёa-z0-9-]+/gi) || [];
  const ttr = movingTTR(allWords);

  const avgWordLenChars =
    allWords.length > 0
      ? Number((allWords.reduce((sum, w) => sum + w.length, 0) / allWords.length).toFixed(1))
      : 6.2;

  const morpho = computeMorphosyntaxStats(donorText, sentences, allWords);

  const charLenK = Math.max(0.5, donorText.length / 1000);
  const emDashes = (donorText.match(/—/g) || []).length;
  const enDashes = (donorText.match(/–/g) || []).length;

  const dashStyle: '—' | '–' = enDashes > emDashes ? '–' : '—';
  const dashesPer1k = Number(((emDashes + enDashes) / charLenK).toFixed(1));
  const colonsPer1k = Number(((donorText.match(/:/g) || []).length / charLenK).toFixed(1));
  const semicolonsPer1k = Number(((donorText.match(/;/g) || []).length / charLenK).toFixed(1));
  const parensPer1k = Number(((donorText.match(/\(/g) || []).length / charLenK).toFixed(1));
  const questionsCount = (donorText.match(/\?/g) || []).length;

  const usesGuillemets =
    (donorText.match(/[«»]/g) || []).length >= (donorText.match(/"/g) || []).length;

  const copulaHits = sentences.filter((s) => COPULA_REGEX.test(s)).length;
  const copulaPer1k = Number((copulaHits / charLenK).toFixed(1));

  const conjStarts = sentences.filter((s) =>
    /^(?:Однако|Но|Ибо|Ведь|Впрочем|Причем|Причём|Кроме того|Тем не менее|Для|Между тем)\b/i.test(
      s.trim()
    )
  ).length;

  const conjunctionStartsPct =
    sentences.length > 0 ? Math.round((conjStarts / sentences.length) * 100) : 0;

  const genitiveHits = sentences.filter((s) => GENITIVE_CHAIN_REGEX.test(s)).length;
  const genitiveChainRate =
    sentences.length > 0 ? Math.round((genitiveHits / sentences.length) * 100) : 0;

  const lower = donorText.toLowerCase();

  const favoriteMarkers = AUTHOR_DISCOURSE_MARKERS.filter((m) => {
    const regex = new RegExp(`(^|[\\s,;(])${m}($|[\\s,;:.!?])`, 'i');
    return regex.test(lower);
  }).slice(0, 10);

  const { autoMinedOpeners, autoMinedConnectors } = mineOpenersAndConnectors(sentences);

  // Determine paragraph closing style across donor paragraphs
  const closings = paragraphSentences
    .map((ps) => ps[ps.length - 1])
    .filter(Boolean);
  const questionClosings = closings.filter((c) => c.trim().endsWith('?')).length;
  const shortClosings = closings.filter((c) => getSentenceWords(c).length <= 9).length;
  const longClosings = closings.filter((c) => getSentenceWords(c).length >= 20).length;
  const paragraphClosingStyle: DonorDNAProfile['paragraphClosingStyle'] =
    questionClosings > 0
      ? 'question'
      : shortClosings >= Math.max(1, Math.ceil(closings.length * 0.45))
        ? 'concise'
        : longClosings >= Math.max(1, Math.ceil(closings.length * 0.5))
          ? 'reflective'
          : 'balanced';

  const sampleCadence = safeLengths.slice(0, 16);

  // Up to 9 structurally diverse exemplars evenly sampled across donor text
  const exemplars: string[] = [];
  if (sentences.length > 0) {
    const pickedSet = new Set<string>();
    const byLen = sentences
      .map((s, idx) => ({ s: s.trim(), len: sentenceWordLists[idx].length, idx }))
      .filter((x) => x.len >= 3 && x.len <= 55);

    const addPick = (item: { s: string } | undefined) => {
      if (item && !pickedSet.has(item.s) && exemplars.length < 9) {
        pickedSet.add(item.s);
        exemplars.push(item.s);
      }
    };

    // Short punch, medium textured, long periodic, question/dash/colon picks across beginning/middle/end
    addPick(byLen.find((x) => x.len >= 4 && x.len <= 9));
    addPick(byLen.find((x) => /[—–:]/.test(x.s) && x.len >= 9 && x.len <= 28));
    addPick(byLen.find((x) => x.s.endsWith('?') || x.s.includes(';')));
    addPick(byLen.find((x) => x.len >= 22 && x.len <= 48));

    const step = Math.max(1, Math.floor(byLen.length / 5));
    for (let i = 0; i < byLen.length && exemplars.length < 9; i += step) {
      addPick(byLen[i]);
    }
    for (const item of byLen) {
      if (exemplars.length >= 9) break;
      addPick(item);
    }
  }

  const donorInv = extractFactInventory(donorText);
  const seenDonorNorm = new Set<string>();
  const donorTopicTerms: string[] = [];
  for (const term of [...donorInv.properNames, ...donorInv.domainTerms]) {
    const norm = term.toLowerCase().replace(/ё/g, 'е');
    if (!seenDonorNorm.has(norm)) {
      seenDonorNorm.add(norm);
      donorTopicTerms.push(term);
    }
    if (donorTopicTerms.length >= 24) break;
  }

  const authorialDeviations = extractAuthorialDeviationProfile(donorText);

  return {
    sentenceCount: sentences.length,
    paragraphCount: Math.max(1, paragraphs.length),
    avgSentPerPara,
    minSentPerPara,
    maxSentPerPara,
    minLen,
    medianLen,
    maxLen,
    avgLen,
    stdDev,
    burstiness,
    shortRatioPct,
    longRatioPct,
    ttr,
    avgWordLenChars,
    dashStyle,
    dashesPer1k,
    colonsPer1k,
    semicolonsPer1k,
    parensPer1k,
    questionsCount,
    copulaPer1k,
    conjunctionStartsPct,
    genitiveChainRate,
    favoriteMarkers,
    usesGuillemets,
    sampleCadence,
    exemplars,
    authorialDeviations,
    lengthHistogramPct,
    cadenceDeltaMean,
    nominalizationPer100: morpho.nominalizationPer100,
    participleClausePerSent: morpho.participleClausePerSent,
    modalParticlePer1k: morpho.modalParticlePer1k,
    firstSecondPersonPer1k: morpho.firstSecondPersonPer1k,
    autoMinedOpeners,
    autoMinedConnectors,
    paragraphClosingStyle,
    donorTopicTerms,
  };
}

export function formatDonorProfileBlock(
  profile: DonorDNAProfile,
  blueprint?: DonorStyleBlueprint | null,
  forbiddenDonorTerms?: string[]
): string {
  const targetMinBurst = Math.max(0, Number((profile.burstiness - 0.06).toFixed(2)));
  const targetMaxBurst = Math.min(0.95, Number((profile.burstiness + 0.14).toFixed(2)));

  const lines: string[] = [
    `[ПАСПОРТ СТИЛЕВОЙ ДНК АВТОРА-ДОНОРА — ПРИОРИТЕТ №1]`,
    `• Ритм предложений: длина от ${profile.minLen} до ${profile.maxLen} слов (медиана: ${profile.medianLen} сл., среднее: ${profile.avgLen} сл., средний перепад соседних фраз: ±${profile.cadenceDeltaMean} сл.).`,
    `• Распределение длин у донора [1–5 | 6–10 | 11–16 | 17–24 | 25+ сл.]: [${profile.lengthHistogramPct.join('% | ')}%]. Образец ритма подряд: [${profile.sampleCadence.join(' > ')}].`,
    `• Прерывистость (Burstiness): ориентир ${targetMinBurst}–${targetMaxBurst}; коротких фраз ?8 слов: ~${profile.shortRatioPct}%, длинных периодов ?22 слов: ~${profile.longRatioPct}%.`,
    `• Архитектура абзацев: в среднем ${profile.avgSentPerPara} предл./абзац (разброс ${profile.minSentPerPara}–${profile.maxSentPerPara}); тип концовки абзаца у донора: ${profile.paragraphClosingStyle === 'concise' ? 'короткий отточенный финал без лишней воды' : profile.paragraphClosingStyle === 'question' ? 'вопросительный или проблематизирующий сдвиг' : profile.paragraphClosingStyle === 'reflective' ? 'развёрнутое завершение' : 'естественная смысловая точка без шаблонного резюме'}.`,
    `• Морфосинтаксис донора: отглагольные существительные (-ение/-ость/-ция) ~${profile.nominalizationPer100}/100 сл., обособленные причастия/деепричастия ~${profile.participleClausePerSent} на фразу, модальные частицы (ведь/же/ли/бы/вряд ли) ~${profile.modalParticlePer1k}/1k, личные местоимения (я/мы/вы) ~${profile.firstSecondPersonPer1k}/1k.`,

    `• Лексика и грамматика: MATTR ~${profile.ttr}, средняя длина слова ~${profile.avgWordLenChars} симв. Зачины с союзами/связками: ~${profile.conjunctionStartsPct}% предложений.`,
    profile.copulaPer1k < 0.3
      ? `• Связка «является/представляет собой»: донор использует её редко (${profile.copulaPer1k}/1k). Предпочитай живые глаголы или тире.`
      : `• Связка «является»: встречается у донора (~${profile.copulaPer1k}/1k). Сохраняй естественный баланс.`,
    profile.favoriteMarkers.length > 0 || profile.autoMinedConnectors.length > 0
      ? `• Живые маркеры и связки донора: ${Array.from(new Set([...profile.favoriteMarkers, ...profile.autoMinedConnectors])).slice(0, 10).join(', ')}. Вплетай только туда, где просит смысл.`
      : `• Не навязывай вводные слова и связки: переходы должны вытекать из смысла.`,
    profile.autoMinedOpeners.length > 0
      ? `• Привычные биграммы зачина у донора: ${profile.autoMinedOpeners.slice(0, 5).map((o) => `«${o}...»`).join(', ')} — ориентируйся на интонацию, не на дословную копирку.`
      : '',
  ].filter(Boolean);

  if (forbiddenDonorTerms && forbiddenDonorTerms.length > 0) {
    lines.push(
      `• АНТИ-УТЕЧКА ТЕМЫ ДОНОРА: категорически ЗАПРЕЩЕНО переносить в итог предметные слова и имена из Текста 1: [${forbiddenDonorTerms.slice(0, 16).join(', ')}].`

  }

  const deviations = profile.authorialDeviations;

  if (deviations && deviations.signals.length > 0) {
    lines.push('');
    lines.push('[ЗАЩИТА УСТОЙЧИВЫХ АВТОРСКИХ ОСОБЕННОСТЕЙ]');

    );

    for (const signal of deviations.signals) {
      const label: Record<AuthorialDeviationKind, string> = {
        genitiveChain: 'цепочки родительных падежей',
        copula: 'связки «является/представляет собой»',
        symmetry: 'конструкции «не только..., но и...»',
        triad: 'перечисления из трёх элементов',
        repetition: 'лексические повторы',
        shortFragment: 'короткие самостоятельные фразы',
        unusualOpening: 'повторяющиеся синтаксические зачины',
      };

      lines.push(
        `• ${label[signal.kind]}: ${signal.count} случаев; ~${signal.per300Words} на 300 слов; уверенность сигнала ${signal.confidence}.`
      );

      if (signal.examples.length > 0) {
        lines.push(`  Примеры из донора: ${signal.examples.slice(0, 2).map((x) => `«${x}»`).join(' | ')}`);
      }
    }

    lines.push(
      '• ВАЖНО: не считай перечисленные особенности автоматическими дефектами. Если аналогичная конструкция подходит по смыслу, не исправляй её ради формальной «гладкости».'
    );
  } else {
    lines.push('');
    lines.push(
      '[ЗАЩИТА АВТОРСКОЙ МАНЕРЫ] В образце недостаточно повторяющихся сигналов для вывода об устойчивых отклонениях. Не объявляй живую особенность речи ошибкой.'
    );
  }

  if (blueprint) {
    lines.push('');
    lines.push(`[КАЧЕСТВЕННЫЙ СТИЛЕВОЙ ЧЕРТЁЖ ДОНОРА (ИЗВЛЕЧЁН ИЗ ТЕКСТА 1)]`);
    lines.push(`• Позиция автора и залог: ${blueprint.voiceAndPerson}`);
    lines.push(`• Лексический регистр и эпоха речи: ${blueprint.lexicalRegister}`);
    lines.push(`• Способ переходов между мыслями: ${blueprint.transitionStyle}`);
    lines.push(`• Привычки синтаксиса донора: ${blueprint.syntaxHabits.join(' | ')}`);

    if (blueprint.structuralTemplates.length > 0) {
      lines.push(
        `• Структурные обороты-скелеты (копируй форму, не тему): ${blueprint.structuralTemplates.join(' | ')}`
      );
    }

    if (blueprint.avoidFeatures.length > 0) {
      lines.push(`• Чего этот автор обычно избегает: ${blueprint.avoidFeatures.join(' | ')}`);
    }
  }

  return lines.join('\n');
}

function getAuthorialSignal(
  profile: DonorDNAProfile | undefined,
  kind: AuthorialDeviationKind
): AuthorialDeviationSignal | undefined {
  return profile?.authorialDeviations?.signals.find((signal) => signal.kind === kind);
}

function hasProtectedAuthorialPattern(
  profile: DonorDNAProfile | undefined,
  kind: AuthorialDeviationKind,
  candidateCount: number
): boolean {
  const signal = getAuthorialSignal(profile, kind);

  if (!signal || signal.confidence < 0.55 || candidateCount <= 0) {
    return false;
  }

  // Не требуется точного совпадения количества конструкций.
  // Достаточно того, что конструкция регулярно встречается у донора.
  return signal.count >= 2 || signal.per300Words >= 0.3;
}

export function diagnoseTextDefects(
  text: string,
  donorProfile?: DonorDNAProfile,
  sourceText?: string
): DiagnosticReport {
  const paragraphs = text
    .split(/\n{2,}/)
    .map((p) => p.trim())
    .filter(Boolean);

  const sentences = splitSentences(text);
  const lengths = sentences.map((s) => getSentenceWords(s).length);
  const totalWords = lengths.reduce((a, b) => a + b, 0) || 1;
  const avgLen = Number((totalWords / (lengths.length || 1)).toFixed(1));

  const variance =
    lengths.reduce((sum, len) => sum + Math.pow(len - avgLen, 2), 0) /
    (lengths.length || 1);

  const stdDev = Math.sqrt(variance);
  const burstiness = avgLen > 0 ? Number((stdDev / avgLen).toFixed(2)) : 0;

  const shortCount = lengths.filter((n) => n <= 8).length;
  const longCount = lengths.filter((n) => n >= 22).length;

  const shortRatioPct =
    lengths.length > 0 ? Math.round((shortCount / lengths.length) * 100) : 0;

  const longRatioPct =
    lengths.length > 0 ? Math.round((longCount / lengths.length) * 100) : 0;

  const words = text.toLowerCase().match(/[а-яёa-z0-9-]+/gi) || [];
  const ttr = movingTTR(words);
  const morpho = computeMorphosyntaxStats(text, sentences, words);
  const lengthHistogramPct = computeLengthHistogram(lengths);

  const charLenK = Math.max(0.5, text.length / 1000);
  const dashesPer1k = Number(((text.match(/[—–]/g) || []).length / charLenK).toFixed(1));

  const issues: DiagnosticIssue[] = [];
  const lowerFull = text.toLowerCase();
  const foundCliches = EXTENDED_AI_CLICHES.filter((c) => lowerFull.includes(c));

  const copulaSentences: number[] = [];
  let triadCount = 0;
  let genitiveCount = 0;
  let symmetryCount = 0;

  sentences.forEach((s, idx) => {
    const ls = s.toLowerCase();
    const sId = idx + 1;

    // 1. Машинные клише.
    const sCliches = EXTENDED_AI_CLICHES.filter((c) => ls.includes(c));

    if (sCliches.length > 0) {
      issues.push({
        type: 'cliche',
        severity: 'critical',
        sentenceIndexes: [sId],
        message: `Предложение №${sId}: обнаружены возможные машинные штампы (${sCliches.map((c) => `«${c}»`).join(', ')}). Проверь, не является ли это точной цитатой или необходимым терминологическим оборотом; если нет, замени конкретной формулировкой.`,
      });
    }

    // 2. Связка «является/представляет собой».
    if (COPULA_REGEX.test(ls)) {
      copulaSentences.push(sId);
    }

    // 3. Симметричная конструкция.
    if (/не только .+ но и /i.test(ls)) {
      symmetryCount++;

      if (!hasProtectedAuthorialPattern(donorProfile, 'symmetry', 1)) {
        issues.push({
          type: 'symmetry',
          severity: 'minor',
          sentenceIndexes: [sId],
          message: `Предложение №${sId}: конструкция «не только ... но и». Это нормативная конструкция, а не ошибка. Пересмотри её только в том случае, если она звучит шаблонно или нарушает естественный ход мысли.`,
        });
      }
    }

    // 4. Триады.
    if (TRIAD_REGEX.test(ls)) {
      triadCount++;

      if (!hasProtectedAuthorialPattern(donorProfile, 'triad', 1)) {
        issues.push({
          type: 'triad',
          severity: 'minor',
          sentenceIndexes: [sId],
          message: `Предложение №${sId}: перечисление из трёх элементов. Само по себе оно не является дефектом. Проверяй только искусственное, формальное или семантически избыточное перечисление; не удаляй содержательно необходимые элементы.`,
        });
      }
    }

    // 5. Цепочки родительных падежей.
    if (GENITIVE_CHAIN_REGEX.test(ls)) {
      genitiveCount++;

      if (!hasProtectedAuthorialPattern(donorProfile, 'genitiveChain', 1)) {
        issues.push({
          type: 'genitiveChain',
          severity: 'minor',
          sentenceIndexes: [sId],
          message: `Предложение №${sId}: обнаружена возможная цепочка родительных падежей. Проверь её читаемость и точность. Не перестраивай автоматически, если цепочка нормативна, терминологически точна или характерна для автора.`,
        });
      }
    }

    // 6. Лексическое эхо однокоренных слов внутри предложения.
    const sStems = getSentenceWords(s)
      .filter((w) => w.length >= 5 && !RUSSIAN_FUNCTION_WORDS.has(w.toLowerCase()))
      .map((w) => ({ raw: w, stem: stemRussianWord(w) }))
      .filter((x) => x.stem.length >= 4);

    const seenInSent = new Map<string, string>();

    for (const item of sStems) {
      const previous = seenInSent.get(item.stem);

      if (previous && previous.toLowerCase() !== item.raw.toLowerCase()) {
        issues.push({
          type: 'lexicalEcho',
          severity: 'minor',
          sentenceIndexes: [sId],
          message: `Предложение №${sId}: возможно, близко повторён корень «${item.raw}» / «${previous}». Не устраняй повтор автоматически: сначала определи, является ли он смысловым, терминологическим или авторским.`,
        });
        break;
      }

      if (!previous) {
        seenInSent.set(item.stem, item.raw);
      }
    }
  });

  // Связка оценивается с учетом реальной частоты у донора.
  const donorCopulaRate = donorProfile?.copulaPer1k ?? 0;
  const copulaThreshold =
    donorProfile && donorCopulaRate >= 0.3
      ? Math.max(3, Math.ceil((donorCopulaRate * text.length) / 1000 * 2.5))
      : 2;

  if (copulaSentences.length >= copulaThreshold) {
    issues.push({
      type: 'copula',
      severity: 'minor',
      sentenceIndexes: copulaSentences,
      message: `Предложения №${copulaSentences.join(', №')}: связка «является / представляет собой» встречается несколько раз. Проверь, нет ли канцелярского нагромождения. Если такая конструкция характерна для донора или нужна по смыслу, сохраняй её.`,
    });
  }

  // 7. Гладкие концовки абзацев.
  let runningSentIdx = 0;

  for (const para of paragraphs) {
    const pSents = splitParagraph(para);

    if (pSents.length > 0) {
      const lastSent = pSents[pSents.length - 1];
      const lastSentNum = runningSentIdx + pSents.length;

      if (SMOOTH_ENDING_REGEX.test(lastSent)) {
        issues.push({
          type: 'smoothEnding',
          severity: 'minor',
          sentenceIndexes: [lastSentNum],
          message: `Предложение №${lastSentNum} (конец абзаца): возможно, присутствует шаблонная обобщающая концовка. Проверь, действительно ли она нужна; не обрывай естественно завершённую мысль.`,
        });
      }

      runningSentIdx += pSents.length;
    }
  }

  // 8. Монотонность длин.
  // Не считаем близкие по длине предложения ошибкой, если такой ритм
  // сам по себе характерен для конкретного донора.
  if (lengths.length >= 4) {
    const donorBurstiness = donorProfile?.burstiness;
    const rhythmIsNaturallyEven =
      donorBurstiness !== undefined && donorBurstiness < 0.28;

    if (!rhythmIsNaturallyEven) {
      for (let i = 0; i <= lengths.length - 3; i++) {
        const slice = lengths.slice(i, i + 3);
        const minW = Math.min(...slice);
        const maxW = Math.max(...slice);

        if (maxW - minW <= 4 && minW >= 9) {
          issues.push({
            type: 'monotony',
            severity: 'minor',
            sentenceIndexes: [i + 1, i + 2, i + 3],
            message: `Предложения №${i + 1}–№${i + 3} близки по длине (${slice.join(', ')} слов). Проверь, не создаёт ли это монотонность. Не дроби и не объединяй предложения, если ритм соответствует донору и смыслу.`,
          });

          i += 2;
        }
      }
    }
  }

  // 9. Повторяющиеся зачины.
  for (let i = 0; i < sentences.length - 1; i++) {
    const patA = getOpenerPattern(sentences[i]);
    const patB = getOpenerPattern(sentences[i + 1]);

    if (patA && patA === patB && !patA.startsWith('W(')) {
      const repeatedOpeningIsDonorHabit =
        donorProfile?.authorialDeviations?.protectionLevel === 'high' &&
        hasProtectedAuthorialPattern(donorProfile, 'unusualOpening', 1);

      if (!repeatedOpeningIsDonorHabit) {
        issues.push({
          type: 'repetition',
          severity: 'minor',
          sentenceIndexes: [i + 1, i + 2],
          message: `Предложения №${i + 1} и №${i + 2} имеют сходный синтаксический зачин (${patA}). Проверь, не звучит ли повтор механически. Если повторение соответствует манере донора, оставь его.`,
        });
      }
    } else {
      const firstA = (getSentenceWords(sentences[i])[0] || '').toLowerCase();
      const firstB = (getSentenceWords(sentences[i + 1])[0] || '').toLowerCase();

      if (firstA && firstA === firstB && firstA.length > 2) {
        const repeatedOpeningIsDonorHabit =
          donorProfile?.authorialDeviations?.protectionLevel === 'high' &&
          hasProtectedAuthorialPattern(donorProfile, 'unusualOpening', 1);

        if (!repeatedOpeningIsDonorHabit) {
          issues.push({
            type: 'repetition',
            severity: 'minor',
            sentenceIndexes: [i + 1, i + 2],
            message: `Предложения №${i + 1} и №${i + 2} начинаются с одного слова («${firstA}»). Проверь, не является ли это случайным повтором. Не меняй зачин, если он характерен для донора.`,
          });
        }
      }
    }
  }

  // 10. Прерывистость.
  const minTargetBurstiness = donorProfile
    ? Math.max(0, Number((donorProfile.burstiness - 0.1).toFixed(2)))
    : 0.35;

  if (sentences.length >= 5 && burstiness < minTargetBurstiness) {
    issues.push({
      type: 'burstiness',
      severity: 'minor',
      sentenceIndexes: [],
      message: `Прерывистость (Burstiness = ${burstiness}) ниже ориентира (${minTargetBurstiness}). Проверь, не получился ли текст слишком ровным. Не добавляй короткие фразы и длинные периоды искусственно: ориентируйся на естественную интонацию донора.`,
    });
  }

  // 11. Дрейф длины предложений относительно донора.
  if (donorProfile && sentences.length >= 5) {
    if (
      donorProfile.shortRatioPct >= 20 &&
      shortRatioPct < donorProfile.shortRatioPct * 0.25
    ) {
      issues.push({
        type: 'donorLengthDrift',
        severity: 'minor',
        sentenceIndexes: [],
        message: `В тексте заметно меньше коротких фраз, чем у донора (${shortRatioPct}% против ${donorProfile.shortRatioPct}%). Проверь ритм, но не дроби предложения механически.`,
      });
    }

    if (
      donorProfile.longRatioPct >= 20 &&
      longRatioPct < donorProfile.longRatioPct * 0.25
    ) {
      issues.push({
        type: 'donorLengthDrift',
        severity: 'minor',
        sentenceIndexes: [],
        message: `В тексте заметно меньше длинных периодов, чем у донора (${longRatioPct}% против ${donorProfile.longRatioPct}%). Проверь, не упрощён ли синтаксис чрезмерно.`,
      });
    }

    if (Math.abs(avgLen - donorProfile.avgLen) >= 9) {
      issues.push({
        type: 'donorLengthDrift',
        severity: 'minor',
        sentenceIndexes: [],
        message: `Средняя длина фразы (${avgLen} слов) отличается от донора (${donorProfile.avgLen} слов). Проверь соответствие ритма, не подгоняя длину механически.`,
      });
    }
  }

  // 12. Пунктуационный почерк.
  if (donorProfile && text.length >= 400) {
    if (
      donorProfile.dashesPer1k >= 3.0 &&
      dashesPer1k < donorProfile.dashesPer1k * 0.35
    ) {
      issues.push({
        type: 'donorPunctuation',
        severity: 'minor',
        sentenceIndexes: [],
        message: `Тире встречаются реже, чем у донора (${dashesPer1k}/1k против ${donorProfile.dashesPer1k}/1k). Проверь пунктуационную интонацию, но не вставляй тире там, где они не нужны.`,
      });
    } else if (dashesPer1k > Math.max(8, donorProfile.dashesPer1k * 2.4)) {
      issues.push({
        type: 'donorPunctuation',
        severity: 'minor',
        sentenceIndexes: [],
        message: `Тире встречаются заметно чаще, чем у донора (${dashesPer1k}/1k против ${donorProfile.dashesPer1k}/1k). Проверь, не появились ли они искусственно.`,
      });
    }
  }

  // 13. Дискурсивные маркеры.
  // Отсутствие маркеров не является дефектом: добавлять их ради статистики нельзя.
  if (donorProfile && donorProfile.favoriteMarkers.length >= 3 && sentences.length >= 5) {
    const usedMarkers = donorProfile.favoriteMarkers.filter((m) =>
      new RegExp(`(^|[\\s,;(])${m}($|[\\s,;:.!?])`, 'i').test(lowerFull)
    );

    if (usedMarkers.length === 0) {
      issues.push({
        type: 'donorMarkers',
        severity: 'minor',
        sentenceIndexes: [],
        message: `В тексте не встретились некоторые характерные связки донора (${donorProfile.favoriteMarkers.slice(0, 5).join(', ')}). Это не требует обязательного исправления: не добавляй маркеры, если они не подходят по смыслу.`,
      });
    }
  }

  // 14. MATTR.
  // Низкий показатель — только повод посмотреть на текст, а не команда
  // заменять точные термины и авторские повторы синонимами.
  if (words.length >= 80 && ttr < 0.5) {
    issues.push({
      type: 'ttr',
      severity: 'minor',
      sentenceIndexes: [],
      message: `Корневое разнообразие MATTR = ${ttr}. Проверь, не вызвано ли оно чрезмерными случайными повторами. Сохраняй термины, смысловые повторы и характерную лексику автора; не подбирай синонимы ради числового показателя.`,
    });
  }

  // 15. Морфосинтаксический перекос (отглагольные существительные и причастные обороты).
  if (words.length >= 35) {
    const nomLimit = donorProfile
      ? Math.max(8.5, donorProfile.nominalizationPer100 * 1.65)
      : 12.0;

    if (morpho.nominalizationPer100 > nomLimit) {
      issues.push({
        type: 'nominalizationOverload',
        severity: 'major',
        sentenceIndexes: [],
        message: `Перегруз отглагольными существительными (-ение/-ость/-ция): ${morpho.nominalizationPer100}/100 слов (у донора ${donorProfile ? donorProfile.nominalizationPer100 : '6–9'}/100). Заменяй канцелярские конструкции с отглагольными существительными на живые глаголы действия.`,
      });
    }

    const partLimit = donorProfile
      ? Math.max(0.75, donorProfile.participleClausePerSent + 0.45)
      : 1.2;

    if (sentences.length >= 3 && morpho.participleClausePerSent > partLimit) {
      issues.push({
        type: 'participleOverload',
        severity: 'minor',
        sentenceIndexes: [],
        message: `Плотность обособленных причастных/деепричастных оборотов (${morpho.participleClausePerSent}/предл.) выше естественного уровня${donorProfile ? ` донора (${donorProfile.participleClausePerSent})` : ''}. Переведи часть оборотов в простые придаточные или глагольные конструкции.`,
      });
    }
  }

  // 16. Сохранность фактов, отрицаний, модальности и анти-утечка темы донора.
  const factsReport: FactVerificationReport = sourceText
    ? verifyFactsPreserved(sourceText, text, donorProfile)
    : {
        preservedPct: 100,
        lostCritical: [],
        lostTerms: [],
        negationDelta: 0,
        lostModals: [],
        donorLeakTerms: [],
      };

  if (factsReport.lostCritical.length > 0) {
    issues.unshift({
      type: 'factLoss',
      severity: 'critical',
      sentenceIndexes: [],
      message: `КРИТИЧНО: возможно, потеряны обязательные факты/имена/числа из исходника: ${factsReport.lostCritical.slice(0, 8).join(', ')}. Проверь исходный материал и верни пропущенное без искажения.`,
    });
  }

  if (factsReport.donorLeakTerms && factsReport.donorLeakTerms.length > 0) {
    const leakSentIds: number[] = [];
    sentences.forEach((s, idx) => {
      const ls = s.toLowerCase().replace(/ё/g, 'е');
      if (
        factsReport.donorLeakTerms!.some(
          (term) => ls.includes(term.toLowerCase().replace(/ё/g, 'е'))
        )
      ) {
        leakSentIds.push(idx + 1);
      }
    });

    issues.unshift({
      type: 'donorTopicLeak',
      severity: 'critical',
      sentenceIndexes: leakSentIds.slice(0, 6),
      message: `КРИТИЧНО (утечка темы донора): в текст попали предметные слова из Текста 1 (${factsReport.donorLeakTerms.slice(0, 8).join(', ')}). Немедленно убери тематическую лексику донора, оставив только материал Текста 2.`,
    });
  }

  if (
    sourceText &&
    sentences.length >= 3 &&
    ((factsReport.negationDelta !== undefined && Math.abs(factsReport.negationDelta) >= 2) ||
      (factsReport.lostModals && factsReport.lostModals.length >= 2))
  ) {
    const modalNote =
      factsReport.lostModals && factsReport.lostModals.length > 0
        ? ` Ослаблены модальные маркеры (${factsReport.lostModals.slice(0, 5).join(', ')}).`
        : '';
    const negNote =
      factsReport.negationDelta !== undefined && Math.abs(factsReport.negationDelta) >= 2
        ? ` Сдвиг числа отрицаний относительно исходника: ${factsReport.negationDelta > 0 ? `потеряно ~${factsReport.negationDelta}` : `добавлено ~${Math.abs(factsReport.negationDelta)}`}.`
        : '';

    issues.push({
      type: 'negationShift',
      severity: 'major',
      sentenceIndexes: [],
      message: `Смысловой сдвиг полярности или модальности:${negNote}${modalNote} Сверь оговорки, ограничения и отрицания с Текстом 2.`,
    });
  }

  // Подсчёт индексов.
  const criticalCount = issues.filter((i) => i.severity === 'critical').length;
  const majorCount = issues.filter((i) => i.severity === 'major').length;
  const minorCount = issues.filter((i) => i.severity === 'minor').length;

  const humanScore = Math.max(
    10,
    Math.min(100, Math.round(100 - criticalCount * 14 - majorCount * 7 - minorCount * 2))
  );

  let dnaMatchScore = humanScore;

  if (donorProfile) {
    const burstDiff = Math.abs(burstiness - donorProfile.burstiness);
    const avgLenDiff = Math.abs(avgLen - donorProfile.avgLen);
    const shortDiff = Math.abs(shortRatioPct - donorProfile.shortRatioPct);
    const longDiff = Math.abs(longRatioPct - donorProfile.longRatioPct);
    const dashDiff = Math.abs(dashesPer1k - donorProfile.dashesPer1k);

    const histL1 = donorProfile.lengthHistogramPct.reduce(
      (sum, pct, idx) => sum + Math.abs(pct - (lengthHistogramPct[idx] ?? 0)),
      0
    );
    const histPenalty = Math.min(8, Math.max(0, (histL1 - 24) * 0.08));

    const nomDiff = Math.max(0, Math.abs(morpho.nominalizationPer100 - donorProfile.nominalizationPer100) - 2.0);
    const partDiff = Math.max(0, Math.abs(morpho.participleClausePerSent - donorProfile.participleClausePerSent) - 0.25);
    const morphoPenalty = Math.min(7, nomDiff * 0.55) + Math.min(5, partDiff * 3.0);

    const markerHits = donorProfile.favoriteMarkers.filter((m) =>
      lowerFull.includes(m)
    ).length;

    // Отсутствие маркеров не штрафуется: их нельзя вставлять ради баллов.
    const markerBonus =
      donorProfile.favoriteMarkers.length > 0
        ? Math.min(5, markerHits * 1.5)
        : 2;

    const penalty =
      burstDiff * 30 +
      Math.min(15, avgLenDiff * 1.1) +
      shortDiff * 0.14 +
      longDiff * 0.12 +
      Math.min(10, dashDiff * 1.1) +
      histPenalty +
      morphoPenalty +
      criticalCount * 8 +
      majorCount * 4;

    dnaMatchScore = Math.max(
      10,
      Math.min(100, Math.round(95 + markerBonus - penalty))
    );
  }

  const factPenalty =
    factsReport.lostCritical.length * 12 +
    (factsReport.donorLeakTerms?.length ?? 0) * 10;

  const overallScore = Math.max(
    5,
    Math.min(
      100,
      Math.round(
        (donorProfile
          ? dnaMatchScore * 0.55 + humanScore * 0.45
          : humanScore) - factPenalty
      )
    )
  );

  const needsRefinement =
    criticalCount > 0 ||
    majorCount > 0 ||
    overallScore < 88 ||
    issues.some((issue) => issue.type === 'factLoss' || issue.type === 'donorTopicLeak');

  const severityOrder: Record<DiagnosticIssue['severity'], number> = {
    critical: 0,
    major: 1,
    minor: 2,
  };
  const prioritizedIssues = [...issues]
    .sort((a, b) => severityOrder[a.severity] - severityOrder[b.severity])
    .slice(0, 12);

  return {
    sentenceCount: sentences.length,
    paragraphCount: paragraphs.length,
    avgLen,
    shortRatioPct,
    longRatioPct,
    burstiness,
    ttr,
    dashesPer1k,
    nominalizationPer100: morpho.nominalizationPer100,
    participleClausePerSent: morpho.participleClausePerSent,
    foundCliches,
    issues: prioritizedIssues,
    humanScore,
    dnaMatchScore,
    overallScore,
    factsReport,
    needsRefinement,
  };
}

export function numberSentencesWithParagraphs(text: string): {
  numberedText: string;
  paragraphs: string[][];
} {
  let n = 0;
  const paragraphs: string[][] = [];
  const numberedParagraphs: string[] = [];

  for (const para of text.split(/\n{2,}/)) {
    const sents = splitParagraph(para);

    if (sents.length === 0) continue;

    paragraphs.push(sents);
    numberedParagraphs.push(sents.map((s) => `[${++n}] ${s}`).join(' '));
  }

  return {
    numberedText: numberedParagraphs.join('\n\n'),
    paragraphs,
  };
}

// Применяет JSON-патчи только к указанным предложениям.
export function applySentencePatches(
  text: string,
  patches: Array<{ sentenceId: number; rewritten: string }>
): string {
  const patchMap = new Map<number, string>();

  for (const p of patches) {
    if (
      p &&
      Number.isInteger(p.sentenceId) &&
      typeof p.rewritten === 'string' &&
      p.rewritten.trim()
    ) {
      const cleaned = p.rewritten.replace(/^\[\d+\]\s*/, '').trim();

      if (cleaned.length >= 2) {
        patchMap.set(p.sentenceId, cleaned);
      }
    }
  }

  if (patchMap.size === 0) return text;

  let n = 0;

  const rebuilt = text
    .split(/\n{2,}/)
    .map((para) => {
      const sents = splitParagraph(para);

      if (sents.length === 0) return para.trim();

      return sents
        .map((s) => {
          n++;
          return patchMap.get(n) || s;
        })
        .join(' ');
    })
    .join('\n\n');

  return rebuilt;
}

function buildAuthorialProtectionReminder(
  donorProfile?: DonorDNAProfile
): string {
  if (!donorProfile?.authorialDeviations) {
    return `
ЗАЩИТА АВТОРСКОГО ПОЧЕРКА:
- Не исправляй необычную грамматику только потому, что она отличается от нейтральной нормы.
- Не удаляй намеренные повторы, тяжеловесные конструкции, короткие фразы и нестандартные зачины автоматически.

  }

  const profile = donorProfile.authorialDeviations;

  const descriptions: Record<AuthorialDeviationKind, string> = {

    unusualOpening: 'необычные синтаксические зачины',
  };

    .map(
      (signal) =>
        `• ${descriptions[signal.kind]}: ${signal.count} случаев в образце, уверенность ${signal.confidence}.`

- Не добавляй подобные особенности искусственно: переноси их только там, где они органичны.

}

export function extractLastSentences(text: string, count = 2): string {
  const sents = splitSentences(text);
  if (sents.length === 0) return '';
  return sents.slice(-Math.max(1, count)).join(' ');
}

  const issuesList = [

function selectRepresentativeText(text: string, maxChars: number): string {
  const normalized = text.trim();

  if (normalized.length <= maxChars) return normalized;

  const chunkSize = Math.floor(maxChars / 3);

  const starts = [
    0,
    Math.max(0, Math.floor((normalized.length - chunkSize) / 2)),
    Math.max(0, normalized.length - chunkSize),
  ];

  const chunks = starts.map((start) =>
    normalized.slice(start, start + chunkSize).trim()
  );

  return chunks.filter(Boolean).join('\n[… фрагмент пропущен …]\n');
}

export function deterministicPolish(
  text: string,
  donorProfile?: DonorDNAProfile
): string {
  let out = text.trim();

  // Удаляем нулевые/невидимые символы, которые иногда встречаются как артефакты.
  out = out.replace(/[\u200B-\u200D\uFEFF\u00AD]/g, '');

  // Убираем вводные реплики модели только перед двоеточием.
  out = out
    .replace(
      /^(?:Вот [^\n:]{0,60}|Итоговый текст|Переписанный (?:текст|вариант)|Результат|Готовый текст)\s*:\s*/i,
      ''
    )
    .trim();

  if (
    (out.startsWith('```') && out.endsWith('```')) ||
    (out.startsWith('"""') && out.endsWith('"""'))
  ) {
    out = out
      .replace(/^```[a-z]*\n?/i, '')
      .replace(/\n?```$/i, '')
      .trim();
  }

  // Снимаем случайный Markdown.
  out = out
    .replace(/^#{1,6}\s+/gm, '')
    .replace(/\*\*([^*\n]{1,120})\*\*/g, '$1');

  // Убираем внешние кавычки только тогда, когда они обрамляют весь результат.
  // Внутренние кавычки сохраняются без нормализации.
  if (
    (out.startsWith('«') && out.endsWith('»') && out.indexOf('»') === out.length - 1) ||
    (out.startsWith('"') && out.endsWith('"') && out.indexOf('"', 1) === out.length - 1)
  ) {
    out = out.slice(1, -1).trim();
  }

  // Удаляем техническую нумерацию предложений.
  out = out
    .replace(/(^|\n)\[\d+\]\s+/g, '$1')
    .replace(/\s\[\d+\]\s+/g, ' ');

  // Не нормализуем все тире в тексте: это может уничтожить авторскую пунктуацию.
  // Меняем только очевидные разделительные тире с пробелами вокруг,
  // если профиль донора явно указывает на преобладающий вариант.
  if (donorProfile) {
    const preferredDash = donorProfile.dashStyle;
    const alternateDash = preferredDash === '—' ? '–' : '—';

    const preferredCount = (out.match(new RegExp(preferredDash, 'g')) || []).length;
    const alternateCount = (out.match(new RegExp(alternateDash, 'g')) || []).length;

    // Не меняем пунктуацию, если оба варианта используются сопоставимо часто.
    // Это защищает авторское смешанное употребление тире.
    if (alternateCount > preferredCount * 2 && alternateCount >= 4) {
      out = out.replace(
        new RegExp(`\\s${alternateDash}\\s`, 'g'),
        ` ${preferredDash} `
      );
    }
  }

  // Кавычки намеренно не нормализуются.
  // «Ёлочки», "лапки" и другие варианты могут быть частью почерка донора.

  out = out
    .split('\n')
    .map((line) => line.replace(/[ \t]{2,}/g, ' ').trimEnd())
    .join('\n')
    .replace(/\n{3,}/g, '\n\n')
    .trim();

  return out;
}
